   204→   204→function toIsoDate(d: Date): string {
   205→   205→  const p = (n: number) => String(n).padStart(2, "0");
   206→   206→  return `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())}`;
   207→   207→}
   208→   208→
   209→   209→export function localToday(): string {
   210→   210→  return toIsoDate(new Date());

   213→   213→function addDaysIso(iso: string, days: number): string {
   214→   214→  const [y, m, d] = iso.split("-").map(Number);
   215→   215→  return toIsoDate(new Date(y, m - 1, d + days));
   216→   216→}

   237→   237→/** "Oct 12, 2026" from an ISO date or full timestamp. */
   238→   238→export function prettyDay(iso: string): string {
   239→   239→  const d = new Date(iso);
   240→   240→  if (Number.isNaN(d.getTime())) return iso;
   241→   241→  return d.toLocaleDateString("en-US", {
   242→   242→    month: "short",
   243→   243→    day: "numeric",
   244→   244→    year: "numeric",
   245→   245→  });
   246→   246→}
   247→   247→
   248→   248→/** "Oct 12, 2026 · 4:26 PM" from an ISO timestamp. */
   249→   249→export function prettyStamp(iso: string): string {
   250→   250→  const d = new Date(iso);
   251→   251→  if (Number.isNaN(d.getTime())) return iso;
   252→   252→  return `${prettyDay(iso)} · ${d.toLocaleTimeString("en-US", {
   253→   253→    hour: "numeric",
   254→   254→    minute: "2-digit",
   255→   255→  })}`;
   256→   256→}
   257→   257→
   258→   258→/** "3 days ago" style relative label (best-effort, local clock). */
   259→   259→export function relativeTime(iso: string): string {
   260→   260→  const then = new Date(iso).getTime();
   261→   261→  if (Number.isNaN(then)) return "";
   262→   262→  const mins = Math.round((Date.now() - then) / 60_000);
   263→   263→  if (mins < 1) return "just now";
   264→   264→  if (mins < 60) return `${mins} min ago`;
   265→   265→  const hours = Math.round(mins / 60);
   266→   266→  if (hours < 24) return `${hours} hr ago`;
   267→   267→  const days = Math.round(hours / 24);
   268→   268→  if (days < 7) return `${days} day${days === 1 ? "" : "s"} ago`;
   269→   269→  const weeks = Math.round(days / 7);
   270→   270→  if (weeks < 5) return `${weeks} wk ago`;
   271→   271→  const months = Math.round(days / 30);
   272→   272→  if (months < 12) return `${months} mo ago`;
   273→   273→  return `${Math.round(months / 12)} yr ago`;
   274→   274→}

   276→   276→/** "Mon 29" chart label from a YYYY-MM-DD date. */
   277→   277→export function chartDayLabel(iso: string): string {
   278→   278→  const [y, m, d] = iso.split("-").map(Number);
   279→   279→  if (!y || !m || !d) return iso;
   280→   280→  const dt = new Date(Date.UTC(y, m - 1, d));
   281→   281→  const weekday = dt.toLocaleDateString("en-US", {
   282→   282→    timeZone: "UTC",
   283→   283→    weekday: "short",
   284→   284→  });
   285→   285→  return `${weekday} ${d}`;
   286→   286→}

   318→   318→/** Shared CSV download — the filename comes from Content-Disposition. */
   319→   319→export async function downloadCsvFile(
   320→   320→  path: string,
   321→   321→  fallbackName: string
   322→   322→): Promise<"ok" | "unauthorized" | "error"> {
   323→   323→  try {
   324→   324→    const res = await fetch(path);
   325→   325→    if (res.status === 401) return "unauthorized";
   326→   326→    if (!res.ok) return "error";
   327→   327→    const blob = await res.blob();
   328→   328→    const disposition = res.headers.get("Content-Disposition") ?? "";
   329→   329→    const filename =
   330→   330→      /filename="([^"]+)"/.exec(disposition)?.[1] ?? fallbackName;
   331→   331→    const url = URL.createObjectURL(blob);
   332→   332→    const a = document.createElement("a");
   333→   333→    a.href = url;
   334→   334→    a.download = filename;
   335→   335→    document.body.appendChild(a);
   336→   336→    a.click();
   337→   337→    a.remove();
   338→   338→    URL.revokeObjectURL(url);
   339→   339→    return "ok";
   340→   340→  } catch {
   341→   341→    return "error";
   342→   342→  }
   343→   343→}

   396→   396→/* ------------------------------------------------------------------ */
   397→   397→/*  useAdminFetch — cookie-auth wrappers for the admin CRUD APIs        */
   398→   398→/*  (same-origin fetch; 401 routes to onUnauthorized so the shell can   */
   399→   399→/*   return to the login gate)                                          */
   400→   400→/* ------------------------------------------------------------------ */
   401→   401→
   402→   402→export function useAdminFetch(onUnauthorized: () => void) {
   403→   403→  const [loading, setLoading] = useState(false);
   404→   404→  const [error, setError] = useState<string | null>(null);
   405→   405→  const onAuthRef = useRef(onUnauthorized);
   406→   406→  onAuthRef.current = onUnauthorized;
   407→   407→
   408→   408→  const handle401 = useCallback((res: Response): boolean => {
   409→   409→    if (res.status === 401) {
   410→   410→      onAuthRef.current();
   411→   411→      return true;
   412→   412→    }
   413→   413→    return false;
   414→   414→  }, []);
   415→   415→
   416→   416→  const get = useCallback(
   417→   417→    async <T,>(path: string): Promise<T | null> => {
   418→   418→      setLoading(true);
   419→   419→      setError(null);
   420→   420→      try {
   421→   421→        const res = await fetch(path);
   422→   422→        if (handle401(res)) return null;
   423→   423→        const data = await res.json().catch(() => null);
   424→   424→        if (!res.ok) {
   425→   425→          throw new Error(
   426→   426→            (data as { error?: string } | null)?.error ?? `Request failed (${res.status}).`
   427→   427→          );
   428→   428→        }
   429→   429→        return data as T;
   430→   430→      } catch (err) {
   431→   431→        setError(err instanceof Error ? err.message : "Request failed.");
   432→   432→        return null;
   433→   433→      } finally {
   434→   434→        setLoading(false);
   435→   435→      }
   436→   436→    },
   437→   437→    [handle401]
   438→   438→  );
   439→   439→
   440→   440→  const post = useCallback(
   441→   441→    async <T,>(
   442→   442→      path: string,
   443→   443→      payload: unknown
   444→   444→    ): Promise<{ ok: boolean; data: T | null; status: number }> => {
   445→   445→      setLoading(true);
   446→   446→      setError(null);
   447→   447→      try {
   448→   448→        const res = await fetch(path, {
   449→   449→          method: "POST",
   450→   450→          headers: { "Content-Type": "application/json" },
   451→   451→          body: JSON.stringify(payload ?? {}),
   452→   452→        });
   453→   453→        if (handle401(res)) return { ok: false, data: null, status: 401 };
   454→   454→        const data = await res.json().catch(() => null);
   455→   455→        if (!res.ok) {
   456→   456→          const message =
   457→   457→            (data as { error?: string } | null)?.error ?? `Request failed (${res.status}).`;
   458→   458→          setError(message);
   459→   459→          return { ok: false, data: data as T | null, status: res.status };
   460→   460→        }
   461→   461→        return { ok: true, data: data as T, status: res.status };
   462→   462→      } catch (err) {
   463→   463→        setError(err instanceof Error ? err.message : "Request failed.");
   464→   464→        return { ok: false, data: null, status: 0 };
   465→   465→      } finally {
   466→   466→        setLoading(false);
   467→   467→      }
   468→   468→    },
   469→   469→    [handle401]
   470→   470→  );
   471→   471→
   472→   472→  const del = useCallback(
   473→   473→    async <T,>(path: string): Promise<{ ok: boolean; data: T | null; status: number }> => {
   474→   474→      setLoading(true);
   475→   475→      setError(null);
   476→   476→      try {
   477→   477→        const res = await fetch(path, { method: "DELETE" });
   478→   478→        if (handle401(res)) return { ok: false, data: null, status: 401 };
   479→   479→        const data = await res.json().catch(() => null);
   480→   480→        if (!res.ok) {
   481→   481→          const message =
   482→   482→            (data as { error?: string } | null)?.error ?? `Request failed (${res.status}).`;
   483→   483→          setError(message);
   484→   484→          return { ok: false, data: data as T | null, status: res.status };
   485→   485→        }
   486→   486→        return { ok: true, data: data as T, status: res.status };
   487→   487→      } catch (err) {
   488→   488→        setError(err instanceof Error ? err.message : "Request failed.");
   489→   489→        return { ok: false, data: null, status: 0 };
   490→   490→      } finally {
   491→   491→        setLoading(false);
   492→   492→      }
   493→   493→    },
   494→   494→    [handle401]
   495→   495→  );
   496→   496→
   497→   497→  const clearError = useCallback(() => setError(null), []);
   498→   498→
   499→   499→  return { get, post, del, loading, error, handle401, clearError };
   500→   500→}

   723→   723→  const [uploading, setUploading] = useState(false);
   724→   724→  const { fail } = useAdminToast();
   725→   725→
   726→   726→  async function uploadFile(file: File) {
   727→   727→    if (uploading) return;
   728→   728→    setUploading(true);
   729→   729→    try {
   730→   730→      const body = new FormData();
   731→   731→      body.append("file", file);

   734→   734→        | { url?: string; error?: string }
   735→   735→        | null;

   742→   742→        throw new Error(data?.error ?? "The upload was rejected by the server.");
   743→   743→      }
   744→   744→      onChange(data.url);
   745→   745→    } catch (err) {
   746→   746→      fail("Upload failed", err instanceof Error ? err.message : "Please try again.");

   748→   748→      setUploading(false);

   779→   779→              </div>
   780→   780→            )}
   781→   781→          </div>

   789→   789→            >
   790→   790→              <X className="h-3.5 w-3.5" aria-hidden="true" />
   791→   791→            </button>
   792→   792→          ) : null}
   793→   793→          {uploading ? (
   794→   794→            <div className="absolute inset-0 flex items-center justify-center rounded-xl bg-white/80 backdrop-blur-sm">

   809→   809→              if (file) void uploadFile(file);
   810→   810→            }}

  1075→  1075→
  1076→  1076→export function WiredStub({
  1077→  1077→  endpoint,
  1078→  1078→  icon: Icon,
  1079→  1079→  title,
  1080→  1080→  hint,
  1081→  1081→  summarize,
  1082→  1082→  onSessionExpired,
  1083→  1083→}: {
  1084→  1084→  endpoint: string;
  1085→  1085→  icon: LucideIcon;
  1086→  1086→  title: string;
  1087→  1087→  hint?: string;
  1088→  1088→  summarize: (data: Record<string, unknown>) => string;
  1089→  1089→  onSessionExpired: () => void;
  1090→  1090→}) {
  1091→  1091→  const [phase, setPhase] = useState<"loading" | "error" | "ready">("loading");
  1092→  1092→  const [summary, setSummary] = useState("");
  1093→  1093→  const [error, setError] = useState("");
  1094→  1094→  const [tick, setTick] = useState(0);
  1095→  1095→  const summarizeRef = useRef(summarize);
  1096→  1096→  summarizeRef.current = summarize;
  1097→  1097→  const expiredRef = useRef(onSessionExpired);
  1098→  1098→  expiredRef.current = onSessionExpired;
  1099→  1099→
  1100→  1100→  useEffect(() => {
  1101→  1101→    let cancelled = false;
  1102→  1102→    setPhase("loading");
  1103→  1103→    (async () => {
  1104→  1104→      try {
  1105→  1105→        const res = await fetch(endpoint);
  1106→  1106→        if (res.status === 401) {
  1107→  1107→          if (!cancelled) expiredRef.current();
  1108→  1108→          return;
  1109→  1109→        }
  1110→  1110→        const data = await res.json().catch(() => null);
  1111→  1111→        if (!res.ok) {
  1112→  1112→          throw new Error(
  1113→  1113→            (data as { error?: string } | null)?.error ?? `Request failed (${res.status}).`
  1114→  1114→          );
  1115→  1115→        }

  1127→  1127→    };
  1128→  1128→  }, [endpoint, tick]);
  1129→  1129→
  1130→  1130→  if (phase === "loading") return <PanelSkeleton />;
  1131→  1131→
  1132→  1132→  if (phase === "error") {
  1133→  1133→    return (
  1134→  1134→      <div className="rounded-2xl border border-rose-100 bg-rose-50 p-8 text-center">

  1141→  1141→          variant="outline"
  1142→  1142→          size="sm"

  1152→  1152→  return (

  1156→  1156→        title="Editor landing here"

  1159→  1159→          "Module wired and authenticated — the full editor for this section arrives with this update."
  1160→  1160→        }

  1163→  1163→        <span className="inline-flex items-center gap-2 rounded-full bg-blue-50 px-4 py-1.5 text-xs font-bold text-brand-navy ring-1 ring-blue-100">

  1168→  1168→    </div>
