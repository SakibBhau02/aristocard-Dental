    81→
    82→export function EstimatorPage() {
    83→  const { navigate } = useRouter();
    84→  const { lang, d } = useLang();
    85→  const bn = lang === "bn";
    86→  const t = d.estimatorPage;
    87→  const [selection, setSelection] = useState<Selection>({});
    88→
    89→  const formatBDT = (n: number) => {
    90→    const s = `৳${n.toLocaleString("en-IN", { maximumFractionDigits: 0 })}`;
    91→    return bn ? toBnDigits(s) : s;
    92→  };
    93→
    94→  /* localized item name */
    95→  const itemName = (item: EstimatorItem) =>
    96→    bn ? (t.items as Record<string, string>)[item.slug] ?? item.name : item.name;
    97→  const itemUnit = (item: EstimatorItem) =>
    98→    bn ? (t.units as Record<string, string>)[item.unit] ?? item.unit : item.unit;
    99→
   100→  usePageMeta({
   101→    title: "Treatment Cost Estimator — Dental Prices in Dhaka (2026)",
   102→    description:
   103→      "Estimate your dental treatment cost in Bangladesh before your visit. Transparent price ranges for implants, braces, Invisalign, root canals, veneers & more at Aristocrat Dental Care, Uttara Dhaka.",
   104→    jsonLd: {
   105→      "@context": "https://schema.org",
   106→      "@type": "WebApplication",
   107→      name: "Dental Treatment Cost Estimator",
   108→      applicationCategory: "HealthApplication",
   109→      operatingSystem: "Web",
   110→      url: "https://aristocratdentalcarebd.com/#/estimator",
   111→      offersCurrency: "BDT",
   112→      description:
   113→        "Interactive estimator for dental treatment costs in Dhaka, Bangladesh.",
   114→    },
   115→  });
   116→
   117→  const totals = useMemo(() => {
   118→    let min = 0;
   119→    let max = 0;
   120→    let count = 0;
   121→    for (const [slug, qty] of Object.entries(selection)) {
   122→      if (qty <= 0) continue;
   123→      const item = estimatorItems.find((i) => i.slug === slug);
   124→      if (!item) continue;
   125→      min += item.min * qty;
   126→      max += item.max * qty;
   127→      count += qty;
   128→    }
   129→    return { min, max, count, isEmpty: count === 0 };
   130→  }, [selection]);
   131→
   132→  /* Mobile: the sticky bottom estimate bar (when visible) lifts the
   133→     floating action buttons via a body class — nothing overlaps. */
   134→  useEffect(() => {
   135→    document.body.classList.toggle("has-mobile-bar", !totals.isEmpty);
   136→    return () => {
   137→      document.body.classList.remove("has-mobile-bar");
   138→    };
   139→  }, [totals.isEmpty]);
   140→
   141→  const scrollToSummary = () => {
   142→    document
   143→      .getElementById("estimate-summary")
   144→      ?.scrollIntoView({ behavior: "smooth", block: "start" });
   145→  };
   146→

   148→    setSelection((prev) => {
   149→      const next = { ...prev };
   150→      if (qty <= 0) delete next[slug];
   151→      else next[slug] = qty;
   152→      return next;
   153→    });
   154→  };
   155→
   156→  /* Deep-link to booking with the estimate pre-filled */
   157→  const bookWithEstimate = () => {
   158→    const entries = Object.entries(selection).filter(([, qty]) => qty > 0);
   159→    const params = new URLSearchParams();
   160→    const primaryService = entries
   161→      .map(([slug]) => serviceByEstimatorSlug[slug])
   162→      .find(Boolean);
   163→    if (primaryService) params.set("service", primaryService);
   164→    const summary = entries
   165→      .map(([slug, qty]) => {
   166→        const item = estimatorItems.find((i) => i.slug === slug);
   167→        return item ? `${item.name}${qty > 1 ? ` ×${qty}` : ""}` : null;
   168→      })
   169→      .filter(Boolean)
   170→      .join(", ");
   171→    params.set(
   172→      "note",
   173→      `Online estimate: ${summary} — est. ৳${totals.min.toLocaleString("en-IN")}–৳${totals.max.toLocaleString("en-IN")}. Please confirm exact pricing at consultation.`
   174→    );
   175→    navigate(`/appointment?${params.toString()}`);
   176→  };

   178→  const grouped = useMemo(() => {
   179→    const groups: Record<string, EstimatorItem[]> = {};
   180→    for (const item of estimatorItems) {
   181→      groups[item.category] ??= [];

   183→    }
   184→    return groups;
   185→  }, []);
   186→
   187→  return (
   188→    <>
   189→      <PageHeader
   190→        eyebrow={t.eyebrow}
   191→        title={t.title}
   192→        description={t.description}
   193→        breadcrumb={[{ label: d.chrome.nav.home, to: "/" }, { label: t.breadcrumb }]}
   194→      />
   195→
   196→      <section className="py-14 sm:py-16" aria-label="Treatment cost estimator tool">
   197→        <div className="mx-auto max-w-7xl px-4 sm:px-6">
   198→          <div className="grid gap-8 lg:grid-cols-[1.35fr_0.65fr]">
   199→            {/* Item picker */}
   200→            <div>

   204→                title={t.stepTitle}
   205→                description={t.stepText}
   206→              />
   207→
   208→              <div className="mt-10 space-y-10">
   209→                {Object.entries(grouped).map(([category, items]) => (
   210→                  <div key={category}>
   211→                    <h3 className="font-heading flex items-center gap-3 text-sm font-bold uppercase tracking-[0.16em] text-brand-blue">
   212→                      <span className="h-px w-8 bg-brand-blue/40" aria-hidden="true" />
   213→                      {d.categories[category as keyof typeof d.categories] ?? category}
   214→                    </h3>
   215→                    <StaggerContainer className="mt-4 grid gap-3 sm:grid-cols-2">
   216→                      {items.map((item) => {
   217→                        const qty = selection[item.slug] ?? 0;
   218→                        const active = qty > 0;
   219→                        return (
   220→                          <StaggerItem key={item.slug}>
   221→                            <div
   222→                              className={cn(
   223→                                "group flex items-center gap-4 rounded-2xl border p-4 transition-all duration-400",
   224→                                active
   225→                                  ? "border-brand-blue/50 bg-blue-50/60 shadow-[0_10px_30px_-14px_rgba(30,125,255,0.4)]"
   226→                                  : "border-blue-100/70 bg-white hover:border-brand-blue/30 hover:shadow-[0_10px_25px_-14px_rgba(30,125,255,0.3)]"
   227→                              )}
   228→                            >
   229→                              <span
   230→                                className={cn(
   231→                                  "inline-flex h-11 w-11 shrink-0 items-center justify-center rounded-xl transition-all duration-400",
   232→                                  active
   233→                                    ? "bg-brand-blue text-white shadow-lg shadow-blue-500/40"
   234→                                    : "bg-blue-50 text-brand-blue ring-1 ring-blue-100 group-hover:scale-105"
   235→                                )}
   236→                              >
   237→                                <DynamicIcon name={item.icon} className="h-5.5 w-5.5" />
   238→                              </span>
   239→
   240→                              <div className="min-w-0 flex-1">
   241→                                <p className="font-heading text-sm font-bold leading-snug text-brand-navy">
   242→                                  {itemName(item)}
   243→                                </p>
   244→                                <p className="mt-0.5 text-xs text-slate-500">
   245→                                  {formatBDT(item.min)}–{formatBDT(item.max)}{" "}
   246→                                  <span className="text-slate-400">{itemUnit(item)}</span>
   247→                                </p>
   248→                              </div>
   249→
   250→                              {/* Quantity stepper */}
   251→                              <div className="flex items-center gap-1.5">
   252→                                <button
   253→                                  type="button"
   254→                                  onClick={() => setQty(item.slug, Math.max(0, qty - item.step))}
   255→                                  disabled={!active}
   256→                                  aria-label={t.removeOne.replace("{item}", itemName(item))}
   257→                                  className="inline-flex h-8 w-8 items-center justify-center rounded-lg bg-white text-brand-navy ring-1 ring-blue-100 transition-all duration-300 hover:bg-blue-100 disabled:opacity-30"
   258→                                >
   259→                                  <Minus className="h-4 w-4" aria-hidden="true" />
   260→                                </button>
   261→                                <motion.span
   262→                                  key={qty}
   263→                                  initial={{ scale: 1.4 }}
   264→                                  animate={{ scale: 1 }}
   265→                                  className={cn(
   266→                                    "w-7 text-center font-heading text-sm font-bold",
   267→                                    active ? "text-brand-blue" : "text-slate-300"
   268→                                  )}
   269→                                >
   270→                                  {bn ? toBnDigits(qty) : qty}
   271→                                </motion.span>
   272→                                <button
   273→                                  type="button"
   274→                                  onClick={() =>
   275→                                    setQty(item.slug, Math.min(item.maxQty, qty + item.step))
   276→                                  }
   277→                                  aria-label={t.addOne.replace("{item}", itemName(item))}
   278→                                  className="inline-flex h-8 w-8 items-center justify-center rounded-lg bg-brand-blue text-white shadow-md shadow-blue-500/30 transition-all duration-300 hover:bg-brand-navy"
   279→                                >
   280→                                  <Plus className="h-4 w-4" aria-hidden="true" />
   281→                                </button>
   282→                              </div>
   283→                            </div>
   284→                          </StaggerItem>
   285→                        );
   286→                      })}
   287→                    </StaggerContainer>
   288→                  </div>
   289→                ))}
   290→              </div>
   291→            </div>
   292→
   293→            {/* Sticky summary (scroll target for the mobile bar) */}
   294→            <div id="estimate-summary" className="scroll-mt-28">
   295→              <Reveal delay={0.1}>
   296→                <div className="sticky top-28">
   297→                  <div className="hero-mesh-dark relative overflow-hidden rounded-[2rem] p-7 shadow-2xl shadow-navy-900/40 sm:p-8">
   298→                    <div className="dot-grid-light pointer-events-none absolute inset-0 opacity-40" aria-hidden="true" />
   299→                    <div className="relative">
   300→                      <div className="flex items-center gap-3">
   301→                        <span className="inline-flex h-11 w-11 items-center justify-center rounded-2xl bg-white/10 text-amber-300 ring-1 ring-white/15">
   302→                          <Calculator className="h-5.5 w-5.5" aria-hidden="true" />
   303→                        </span>
   304→                        <div>
   305→                          <h2 className="font-heading text-lg font-bold text-white">
   306→                            {t.totalTitle}
   307→                          </h2>
   308→                          <p className="text-xs text-blue-200/80">
   309→                            {totals.count > 0
   310→                              ? (totals.count > 1 ? t.itemsSelected : t.itemSelected).replace(
   311→                                  "{count}",
   312→                                  bn ? toBnDigits(totals.count) : String(totals.count)
   313→                                )
   314→                              : t.selectToBegin}
   315→                          </p>
   316→                        </div>
   317→                      </div>
   318→
   319→                      <div className="mt-7" aria-live="polite">
   320→                        <AnimatePresence mode="wait">
   321→                          {totals.isEmpty ? (
   322→                            <motion.div
   323→                              key="empty"
   324→                              initial={{ opacity: 0, y: 10 }}
   325→                              animate={{ opacity: 1, y: 0 }}
   326→                              exit={{ opacity: 0, y: -10 }}
   327→                              className="rounded-2xl border border-dashed border-white/25 p-6 text-center"
   328→                            >
   329→                              <Wallet className="mx-auto h-8 w-8 text-blue-300/60" aria-hidden="true" />
   330→                              <p className="mt-3 text-sm text-blue-200/80">
   331→                                {t.emptyText}
   332→                              </p>
   333→                            </motion.div>
   334→                          ) : (
   335→                            <motion.div
   336→                              key="total"
   337→                              initial={{ opacity: 0, y: 10 }}
   338→                              animate={{ opacity: 1, y: 0 }}
   339→                              exit={{ opacity: 0, y: -10 }}
   340→                              className="rounded-2xl bg-white/8 p-6 ring-1 ring-white/12"
   341→                            >
   342→                              <p className="text-center font-heading text-3xl font-bold text-white sm:text-[2.15rem]">
   343→                                {formatBDT(totals.min)}
   344→                                <span className="mx-1 text-blue-300/70">–</span>
   345→                                {formatBDT(totals.max)}
   346→                              </p>
   347→                              <p className="mt-2 text-center text-xs text-blue-200/80">
   348→                                {t.rangeNote}
   349→                              </p>
   350→                            </motion.div>
   351→                          )}
   352→                        </AnimatePresence>
   353→                      </div>
   354→
   355→                      {/* Selected items list */}
   356→                      {!totals.isEmpty && (
   357→                        <ul className="scroll-area mt-6 max-h-48 space-y-2 overflow-y-auto pr-1">
   358→                          {Object.entries(selection).map(([slug, qty]) => {
   359→                            const item = estimatorItems.find((i) => i.slug === slug);
   360→                            if (!item) return null;
   361→                            return (
   362→                              <li
   363→                                key={slug}
   364→                                className="flex items-center justify-between gap-3 rounded-xl bg-white/8 px-4 py-2.5 text-sm ring-1 ring-white/10"
   365→                              >
   366→                                <span className="min-w-0 flex-1 truncate text-blue-100/90">
   367→                                  {itemName(item)}
   368→                                  {qty > 1 && (
   369→                                    <span className="ml-1.5 rounded-md bg-white/15 px-1.5 py-0.5 text-[10px] font-bold text-white">
   370→                                      ×{bn ? toBnDigits(qty) : qty}
   371→                                    </span>
   372→                                  )}
   373→                                </span>
   374→                                <span className="shrink-0 font-semibold text-white">
   375→                                  {formatBDT(item.min * qty)}–{formatBDT(item.max * qty)}
   376→                                </span>
   377→                                <button
   378→                                  type="button"
   379→                                  onClick={() => setQty(slug, 0)}
   380→                                  aria-label={t.removeItem.replace("{item}", itemName(item))}
   381→                                  className="inline-flex h-7 w-7 shrink-0 items-center justify-center rounded-lg text-blue-200/60 transition-colors hover:bg-rose-500/20 hover:text-rose-300"
   382→                                >
   383→                                  <Trash2 className="h-3.5 w-3.5" aria-hidden="true" />
   384→                                </button>
   385→                              </li>
   386→                            );
   387→                          })}
   388→                        </ul>
   389→                      )}
   390→
   391→                      <div className="mt-6 space-y-3">
   392→                        <Button
   393→                          asChild={!totals.isEmpty ? false : true}
   394→                          onClick={totals.isEmpty ? undefined : bookWithEstimate}
   395→                          className="w-full rounded-2xl bg-white py-4 text-sm font-bold text-brand-navy shadow-xl transition-all duration-300 hover:-translate-y-0.5 hover:shadow-2xl"
   396→                        >
   397→                          {totals.isEmpty ? (
   398→                            <a
   399→                              href="#/appointment"
   400→                              onClick={(e) => {
   401→                                e.preventDefault();
   402→                                navigate("/appointment");
   403→                              }}
   404→                              className="inline-flex items-center justify-center gap-2"
   405→                            >
   406→                              <CalendarCheck className="h-4.5 w-4.5" aria-hidden="true" />
   407→                              {t.book}
   408→                            </a>
   409→                          ) : (
   410→                            <span className="inline-flex items-center justify-center gap-2">
   411→                              <CalendarCheck className="h-4.5 w-4.5" aria-hidden="true" />
   412→                              {t.bookWithEstimate}
   413→                            </span>
   414→                          )}
   415→                        </Button>
   416→                        {!totals.isEmpty && (
   417→                          <>
   418→                            <button
   419→                              type="button"
   420→                              onClick={() => window.print()}
   421→                              className="inline-flex w-full items-center justify-center gap-2 rounded-2xl bg-white/10 py-3 text-xs font-semibold text-white ring-1 ring-white/20 transition-all duration-300 hover:bg-white/15"
   422→                            >
   423→                              <Printer className="h-3.5 w-3.5" aria-hidden="true" />
   424→                              {t.print}
   425→                            </button>
   426→                            <button
   427→                              type="button"
   428→                              onClick={() => setSelection({})}
   429→                              className="inline-flex w-full items-center justify-center gap-2 rounded-2xl py-3 text-xs font-semibold text-blue-200/80 ring-1 ring-white/20 transition-all duration-300 hover:bg-white/10 hover:text-white"
   430→                            >
   431→                              <RotateCcw className="h-3.5 w-3.5" aria-hidden="true" />
   432→                              {t.reset}
   433→                            </button>
   434→                          </>
   435→                        )}
   436→                      </div>
   437→                    </div>
   438→                  </div>
   439→
   440→                  {/* assurance card */}
   441→                  <div className="mt-4 rounded-3xl border border-blue-100/80 bg-white p-5">
   442→                    <h3 className="flex items-center gap-2 font-heading text-sm font-bold text-brand-navy">
   443→                      <BadgePercent className="h-4.5 w-4.5 text-brand-blue" aria-hidden="true" />
   444→                      {t.whyRange}
   445→                      <Receipt className="ml-auto h-4 w-4 text-slate-300" aria-hidden="true" />
   446→                    </h3>
   447→                    <p className="mt-2.5 text-xs leading-relaxed text-slate-600">
   448→                      {t.whyRangeText}{" "}
   449→                      <span className="font-semibold text-brand-navy">

   455→              </Reveal>

   457→          </div>
   458→
