    45→function localizeReadTime(readTime: string, bn: boolean, minRead: string) {

   401→function ReadingProgress() {
   402→  const [progress, setProgress] = useState(0);
   403→
   404→  useEffect(() => {
   405→    let raf = 0;
   406→    const update = () => {
   407→      raf = 0;
   408→      const el = document.documentElement;
   409→      const total = el.scrollHeight - el.clientHeight;
   410→      setProgress(total > 0 ? Math.min(1, el.scrollTop / total) : 0);
   411→    };
   412→    const onScroll = () => {
   413→      if (!raf) raf = requestAnimationFrame(update);
   414→    };
   415→    update();
   416→    window.addEventListener("scroll", onScroll, { passive: true });
   417→    window.addEventListener("resize", onScroll, { passive: true });
   418→    return () => {
   419→      window.removeEventListener("scroll", onScroll);
   420→      window.removeEventListener("resize", onScroll);
   421→      if (raf) cancelAnimationFrame(raf);
   422→    };
   423→  }, []);
   424→
   425→  return (
   426→    <div
   427→      className="reading-progress"
   428→      style={{ transform: `scaleX(${progress})` }}
   429→      aria-hidden="true"
   430→    />
   431→  );
   432→}

   434→/* Active-section tracking for the article table of contents */
   435→function useActiveSection(count: number) {
   436→  const [active, setActive] = useState(0);
   437→
   438→  useEffect(() => {
   439→    if (count < 2) return;
   440→    const headings = Array.from({ length: count }, (_, i) =>
   441→      document.getElementById(`section-${i}`)
   442→    );
   443→    const visible = headings.filter(Boolean) as HTMLElement[];
   444→    if (visible.length === 0) return;
   445→    const observer = new IntersectionObserver(
   446→      (entries) => {
   447→        entries.forEach((entry) => {
   448→          if (entry.isIntersecting) {
   449→            const idx = Number(entry.target.id.replace("section-", ""));
   450→            setActive(idx);
   451→          }
   452→        });
   453→      },
   454→      { rootMargin: "-25% 0px -65% 0px", threshold: 0 }
   455→    );
   456→    visible.forEach((h) => observer.observe(h));
   457→    return () => observer.disconnect();
   458→  }, [count]);
   459→  return [active, setActive] as const;
   460→}
