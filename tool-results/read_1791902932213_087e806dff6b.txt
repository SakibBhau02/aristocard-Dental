    52→/* "8 min read" → "৮ মিনিট পাঠ" when Bangla */
    53→function localizeReadTime(readTime: string, bn: boolean, minRead: string) {
    54→  if (!bn) return readTime;
    55→  const n = readTime.match(/\d+/)?.[0] ?? "";
    56→  return n ? `${toBnDigits(n)} ${minRead}` : readTime;
    57→}
    58→
    59→/* "Dr. Shazia Khanam" → "SK" */
    60→function initialsOf(name: string) {
    61→  return name
    62→    .replace("Dr. ", "")
    63→    .split(" ")
    64→    .map((n) => n[0])
    65→    .slice(0, 2)
    66→    .join("");
    67→}
    68→
    69→/* Localized post date (bn-BD renders Bangla digits + month names) */
    70→function formatPostDate(
    71→  date: string,
    72→  bn: boolean,
    73→  opts?: Intl.DateTimeFormatOptions
    74→) {
    75→  return new Date(date).toLocaleDateString(
    76→    bn ? "bn-BD" : "en-GB",
    77→    opts ?? { day: "numeric", month: "short", year: "numeric" }
    78→  );
    79→}
    80→
    81→function localizeCount(bn: boolean, v: number) {
    82→  return bn ? toBnDigits(String(v)) : String(v);
    83→}
    84→
    85→/* Bangla conjunct-insensitive comparison aid: removes hasanta (U+09CD),
    86→   ZWNJ and ZWJ so spelling variants like ইম্প্ল্যান্ট / ইমপ্ল্যান্ট match. */
    87→function stripBnMarks(s: string) {
    88→  return s.replace(/[\u09cd\u200c\u200d]/g, "");
    89→}

   211→function SidebarNewsletter() {
   212→  const { lang, d } = useLang();
   213→  const bn = lang === "bn";
   214→  const t = d.blogPage;
   215→  const [email, setEmail] = useState("");
   216→  const [state, setState] = useState<"idle" | "sending" | "done" | "error">(
   217→    "idle"
   218→  );
   219→
   220→  const submit = async (e: React.FormEvent) => {
   221→    e.preventDefault();
   222→    if (state === "sending") return;
   223→    setState("sending");
   224→    try {
   225→      const res = await fetch("/api/newsletter", {
   226→        method: "POST",
   227→        headers: { "Content-Type": "application/json" },
   228→        body: JSON.stringify({ email }),
   229→      });
   230→      setState(res.ok ? "done" : "error");
   231→    } catch {
   232→      setState("error");
   233→    }
   234→  };
   235→
   236→  return (
   237→    <div className="rounded-3xl border border-blue-100/80 bg-ice p-5">
   238→      <span className="inline-flex h-9 w-9 items-center justify-center rounded-full bg-white text-brand-blue ring-1 ring-blue-100">
   239→        <Mail className="h-4 w-4" aria-hidden="true" />
   240→      </span>
   241→      <h2 className="font-heading mt-3 text-base font-bold text-brand-navy">
   242→        {t.newsletterTitle}
   243→      </h2>
   244→      <p className="mt-1.5 text-xs leading-relaxed text-slate-500">
   245→        {t.newsletterText}
   246→      </p>
   247→      {state === "done" ? (
   248→        <div
   249→          className="mt-4 rounded-2xl bg-emerald-50 p-4 ring-1 ring-emerald-200"
   250→          role="status"
   251→        >
   252→          <p className="flex items-center gap-2 text-sm font-bold text-emerald-700">
   253→            <CheckCircle2 className="h-4 w-4 shrink-0" aria-hidden="true" />
   254→            {t.newsletterDone}
   255→          </p>
   256→          <p className="mt-1 text-xs leading-relaxed text-emerald-600">
   257→            {t.newsletterDoneDesc}
   258→          </p>
   259→        </div>
   260→      ) : (
   261→        <form onSubmit={submit} className="mt-4 flex flex-col gap-2">
   262→          <Input
   263→            type="email"
   264→            required
   265→            value={email}
   266→            onChange={(e) => {
   267→              setEmail(e.target.value);
   268→              if (state === "error") setState("idle");
   269→            }}
   270→            placeholder={t.newsletterPlaceholder}
   271→            aria-label={t.newsletterPlaceholder}
   272→            className="h-11 rounded-full border-blue-200/80 bg-white text-sm shadow-sm placeholder:text-slate-400 focus-visible:border-brand-blue"
   273→          />
   274→          <button
   275→            type="submit"
   276→            disabled={state === "sending"}
   277→            className="inline-flex h-11 items-center justify-center rounded-full bg-brand-navy text-xs font-bold text-white shadow-lg shadow-navy-900/20 transition-all duration-300 hover:bg-brand-blue focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-brand-blue focus-visible:ring-offset-2 disabled:opacity-60"
   278→          >
   279→            {state === "sending" ? "…" : t.newsletterCta}
   280→          </button>
   281→          {state === "error" && (
   282→            <p className="text-xs text-red-500" role="alert">
   283→              {t.newsletterFail} — {t.newsletterFailDesc}
   284→            </p>
   285→          )}
   286→        </form>
   287→      )}
   288→    </div>
   289→  );
   290→}

   296→const PAGE_SIZE_INITIAL = 9;
   297→const PAGE_SIZE_MORE = 6;
   298→
   299→export function BlogPage() {
   300→  const [category, setCategory] = useState("All");
   301→  const { query } = useRouter();
   302→  const { lang, d } = useLang();
   303→  const bn = lang === "bn";
   304→  const t = d.blogPage;
   305→  const { posts } = useBundle();
   306→
   307→  usePageMeta({
   308→    title: "Dental Health Blog — Expert Advice from Uttara's Trusted Dentists",
   309→    description:
   310→      "Evidence-based dental health guides from Aristocrat Dental Care specialists: checkup frequency, root canal facts, Invisalign vs braces, kids' dental care & more.",
   311→    jsonLd: {
   312→      "@context": "https://schema.org",
   313→      "@type": "Blog",
   314→      name: "Aristocrat Dental Care Health Blog",
   315→      url: "https://aristocratdentalcarebd.com/blog",
   316→      publisher: { "@type": "Organization", name: "Aristocrat Dental Care & Orthodontic Center" },
   317→    },
   318→  });
   319→
   320→  /* ?q= deep-link prefill (tag chips on article pages link here).
   321→     The typed value is keyed against the URL query so navigating to
   322→     /blog?q=… adopts it instantly, while typing never rewrites the URL. */
   323→  const qParam = query.get("q") ?? "";
   324→  const [searchState, setSearchState] = useState({ q: qParam, value: qParam });
   325→  const search = searchState.q === qParam ? searchState.value : qParam;
   326→  const setSearch = (v: string) => setSearchState({ q: qParam, value: v });
   327→
   328→  const searching = search.trim().length > 0;
   329→
   330→  /* Category chips — derived from the live posts (same as the static list) */
   331→  const blogCategories = useMemo(
   332→    () => ["All", ...Array.from(new Set(posts.map((p) => p.category)))],
   333→    [posts]
   334→  );
   335→
   336→  /* Category filter combined with text search (EN + BN content + tags).
   337→     Bangla conjuncts can be typed several ways (ইম্প্ল্যান্ট vs ইমপ্ল্যান্ট —
   338→     hasanta placement varies), so BN fields are matched with virama
   339→     (U+09CD) and joiners stripped from BOTH sides; English matching stays
   340→     an exact substring. An exact hit always implies a loose hit, so the
   341→     BN checks collapse into the loose comparison. */
   342→  const filtered = useMemo(() => {
   343→    const q = search.trim().toLowerCase();
   344→    const qLoose = stripBnMarks(q);
   345→    return posts.filter((p) => {
   346→      const matchCategory = category === "All" || p.category === category;
   347→      const matchSearch =
   348→        !q ||
   349→        p.title.toLowerCase().includes(q) ||
   350→        p.excerpt.toLowerCase().includes(q) ||
   351→        p.category.toLowerCase().includes(q) ||
   352→        p.tags.some((tag) => tag.toLowerCase().includes(q)) ||
   353→        (qLoose.length > 0 &&
   354→          ((p.titleBn
   355→            ? stripBnMarks(p.titleBn.toLowerCase()).includes(qLoose)
   356→            : false) ||
   357→            (p.excerptBn
   358→              ? stripBnMarks(p.excerptBn.toLowerCase()).includes(qLoose)
   359→              : false)));
   360→      return matchCategory && matchSearch;
   361→    });
   362→  }, [posts, category, search]);
   363→
   364→  /* pagination — derived against the active filter key so "load more"
   365→     resets automatically whenever the category or search changes */
   366→  const filterKey = `${category}|${search}`;
   367→  const [pageState, setPageState] = useState({
   368→    key: filterKey,
   369→    visible: PAGE_SIZE_INITIAL,
   370→  });
   371→  const visible =
   372→    pageState.key === filterKey ? pageState.visible : PAGE_SIZE_INITIAL;
   373→  const showMore = () =>
   374→    setPageState({ key: filterKey, visible: visible + PAGE_SIZE_MORE });
   375→
   376→  const featured = posts.find((p) => p.featured) ?? posts[0];
   377→  /* The large featured card appears only on the unfiltered view; while
   378→     searching or inside a category every match lives in the grid (so the
   379→     visible count always matches the "Showing X articles" text). */
   380→  const showFeatured = !searching && category === "All" && Boolean(featured);
   381→  const rest = showFeatured
   382→    ? filtered.filter((p) => p.slug !== featured.slug)
   383→    : filtered;
   384→  const shown = rest.slice(0, visible);
   385→  const hasMore = rest.length > visible;
   386→
   387→  /* stats strip — computed from the live bundle */
   388→  const stats = useMemo(
   389→    () => ({
   390→      guides: posts.length,
   391→      topics: new Set(posts.map((p) => p.category)).size,
   392→      authors: new Set(posts.map((p) => p.author)).size,
   393→    }),
   394→    [posts]
   395→  );
   396→
   397→  /* sidebar "latest articles" — 5 most recent by publish date */
   398→  const latestPosts = useMemo(
   399→    () => [...posts].sort((a, b) => b.date.localeCompare(a.date)).slice(0, 5),
   400→    [posts]
   401→  );
   402→
   403→  const resultCountText = t.searchResults.replace(
   404→    "{count}",
   405→    localizeCount(bn, rest.length)
   406→  );
   407→  const showingOfText = t.showingOf
   408→    .replace("{shown}", localizeCount(bn, shown.length))
   409→    .replace("{total}", localizeCount(bn, rest.length));
   410→
   411→  const clearFilters = () => {
   412→    setSearch("");
   413→    setCategory("All");
   414→  };
   415→
   416→  const featuredTitle =
   417→    bn && featured?.titleBn ? featured.titleBn : featured?.title ?? "";
   418→
   419→  return (
   420→    <>
   421→      <PageHeader
   422→        eyebrow={t.eyebrow}
   423→        title={t.title}
   424→        description={t.description}
   425→        breadcrumb={[{ label: d.chrome.nav.home, to: "/" }, { label: d.chrome.nav.blog }]}
   426→      />
   427→
   428→      {/* Stats strip — floats over the hero's lower edge */}
   429→      <section className="relative z-10 mx-auto -mt-9 max-w-4xl px-4 sm:-mt-12 sm:px-6">
   430→        <Reveal>
   431→          <dl className="grid grid-cols-3 divide-x divide-blue-100 rounded-3xl border border-blue-100/80 bg-white/95 p-4 shadow-[0_18px_50px_-24px_rgba(0,32,96,0.28)] backdrop-blur sm:p-6">
   432→            <div className="flex flex-col items-center gap-1 px-2 text-center">
   433→              <dt className="flex items-center gap-1.5 text-[11px] font-medium uppercase tracking-wider text-slate-400">
   434→                <BookOpen className="h-3.5 w-3.5 text-brand-blue" aria-hidden="true" />
   435→                {t.statsGuides}
   436→              </dt>
   437→              <dd className="font-heading text-2xl font-bold tabular-nums text-brand-navy sm:text-3xl">
   438→                {localizeCount(bn, stats.guides)}
   439→              </dd>
   440→            </div>
   441→            <div className="flex flex-col items-center gap-1 px-2 text-center">
   442→              <dt className="flex items-center gap-1.5 text-[11px] font-medium uppercase tracking-wider text-slate-400">
   443→                <LayoutGrid className="h-3.5 w-3.5 text-brand-blue" aria-hidden="true" />
   444→                {t.statsTopics}
   445→              </dt>
   446→              <dd className="font-heading text-2xl font-bold tabular-nums text-brand-navy sm:text-3xl">
   447→                {localizeCount(bn, stats.topics)}
   448→              </dd>
   449→            </div>
   450→            <div className="flex flex-col items-center gap-1 px-2 text-center">
   451→              <dt className="flex items-center gap-1.5 text-[11px] font-medium uppercase tracking-wider text-slate-400">
   452→                <Users className="h-3.5 w-3.5 text-brand-blue" aria-hidden="true" />

   459→          </dl>
   460→        </Reveal>
   461→      </section>
   462→
   463→      {/* Featured post — large image card with gradient overlay (hidden while searching) */}
   464→      {showFeatured && (
   465→        <section className="py-12 sm:py-16" aria-label="Featured article">
   466→          <div className="mx-auto max-w-7xl px-4 sm:px-6">

   472→              >
   473→                {featured.image && (
   474→                  <Image
   475→                    src={featured.image}
   476→                    alt=""
   477→                    fill
   478→                    priority
   479→                    sizes="(max-width: 1280px) 100vw, 1280px"

   482→                )}
   483→                <span
   484→                  className="absolute inset-0 bg-gradient-to-t from-brand-navy via-brand-navy/85 to-brand-navy/45 sm:bg-gradient-to-r sm:from-brand-navy sm:via-brand-navy/90 sm:to-brand-navy/35"
   485→                  aria-hidden="true"

   847→/*  Reading progress bar — top of viewport while reading an article    */
   848→/* ------------------------------------------------------------------ */
   849→
   850→function ReadingProgress() {
   851→  const [progress, setProgress] = useState(0);
   852→
   853→  useEffect(() => {
   854→    let raf = 0;
   855→    const update = () => {
   856→      raf = 0;
   857→      const el = document.documentElement;
   858→      const total = el.scrollHeight - el.clientHeight;
   859→      setProgress(total > 0 ? Math.min(1, el.scrollTop / total) : 0);
   860→    };
   861→    const onScroll = () => {
   862→      if (!raf) raf = requestAnimationFrame(update);
   863→    };
   864→    update();
   865→    window.addEventListener("scroll", onScroll, { passive: true });
   866→    window.addEventListener("resize", onScroll, { passive: true });
   867→    return () => {
   868→      window.removeEventListener("scroll", onScroll);
   869→      window.removeEventListener("resize", onScroll);
   870→      if (raf) cancelAnimationFrame(raf);
   871→    };
   872→  }, []);
   873→
   874→  return (
   875→    <div
   876→      className="reading-progress"
   877→      style={{
   878→        transform: `scaleX(${progress})`,
   879→        /* Lift ABOVE the site-wide ScrollProgress (z-[70]) so the bar a
   880→           reader sees on an article page is the article reading bar; the
   881→           navy→blue→gold ramp matches the brand gradient family. */
   882→        zIndex: 80,
   883→        backgroundImage:
   884→          "linear-gradient(90deg, #012c5f 0%, #1e7dff 60%, #f59e0b 100%)",
   885→      }}
   886→      aria-hidden="true"
   887→    />
   888→  );
   889→}

   891→/* Active-section tracking for the article table of contents.
   892→   `routeKey` (the post slug) re-runs the observer when navigating
   893→   between articles that happen to have the same section count; the
   894→   active index is DERIVED against the key so it resets on navigation
   895→   without setState-in-effect. */
   896→function useActiveSection(count: number, routeKey: string) {
   897→  const [state, setState] = useState({ key: routeKey, active: 0 });
   898→  const active = state.key === routeKey ? state.active : 0;
   899→
   900→  useEffect(() => {
   901→    if (count < 2) return;
   902→    const headings = Array.from({ length: count }, (_, i) =>
   903→      document.getElementById(`section-${i}`)
   904→    );
   905→    const visible = headings.filter(Boolean) as HTMLElement[];
   906→    if (visible.length === 0) return;
   907→    const observer = new IntersectionObserver(
   908→      (entries) => {
   909→        entries.forEach((entry) => {
   910→          if (entry.isIntersecting) {
   911→            const idx = Number(entry.target.id.replace("section-", ""));
   912→            setState({ key: routeKey, active: idx });
   913→          }
   914→        });
   915→      },
   916→      { rootMargin: "-25% 0px -65% 0px", threshold: 0 }
   917→    );
   918→    visible.forEach((h) => observer.observe(h));
   919→    return () => observer.disconnect();
   920→  }, [count, routeKey]);
   921→  return [active, (i: number) => setState({ key: routeKey, active: i })] as const;
   922→}
