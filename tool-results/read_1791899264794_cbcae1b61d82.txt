   296→/* Defensive healer for About blobs saved before a field existed.
   297→   deepMerge replaces ARRAYS wholesale, so an old blob's timeline items
   298→   keep lacking year/icon even after merging — refill per-index from
   299→   the defaults so the public bundle is always complete. */
   300→export function normalizeAboutContent(about: AboutContent): AboutContent {
   301→  const defaults = defaultAboutContent();
   302→  const obj = (v: unknown): Record<string, unknown> =>
   303→    v !== null && typeof v === "object" ? (v as Record<string, unknown>) : {};
   304→  const timeline = (Array.isArray(about.timeline) ? about.timeline : defaults.timeline).map(
   305→    (t, i) => {
   306→      const raw = obj(t);
   307→      return {
   308→        year: str(raw.year) || defaults.timeline[i]?.year || "2024",
   309→        icon: str(raw.icon) || defaults.timeline[i]?.icon || "Sparkles",
   310→        title: str(raw.title),
   311→        titleBn: ostr(raw.titleBn),
   312→        description: str(raw.description),
   313→        descriptionBn: ostr(raw.descriptionBn),
   314→      };
   315→    }
   316→  );
   317→  const storyStats = Array.isArray(about.storyStats)
   318→    ? about.storyStats.map((s) => {
   319→        const raw = obj(s);
   320→        return {
   321→          value: str(raw.value),
   322→          label: str(raw.label),
   323→          labelBn: ostr(raw.labelBn),
   324→        };
   325→      })
   326→    : defaults.storyStats;
   327→  return {
   328→    ...about,
   329→    timeline,
   330→    storyStats,
   331→    founderName: str(about.founderName).trim() || defaults.founderName,
   332→  };
   333→}

   396→/* ------------------------------------------------------------------ */
   397→/*  Deep merge — DB sparse overrides over defaults                     */
   398→/* ------------------------------------------------------------------ */
   399→
   400→export function deepMerge<T>(base: T, override: unknown): T {
   401→  if (override === null || override === undefined) return base;
   402→  if (Array.isArray(base) || Array.isArray(override)) {
   403→    /* Arrays are replaced wholesale (never merged item-by-item). */
   404→    return (Array.isArray(override) ? override : base) as T;
   405→  }
   406→  if (typeof base === "object" && typeof override === "object" && base !== null) {
   407→    const out: Record<string, unknown> = { ...(base as Record<string, unknown>) };
   408→    for (const [k, v] of Object.entries(override as Record<string, unknown>)) {
   409→      const b = (base as Record<string, unknown>)[k];
   410→      out[k] = b !== undefined && b !== null && typeof b === "object" && !Array.isArray(b) && v !== null && typeof v === "object" && !Array.isArray(v)
   411→        ? deepMerge(b, v)
   412→        : v;
   413→    }
   414→    return out as T;
   415→  }
   416→  return (override as T) ?? base;
   417→}

   419→export function featuredServices(list: Service[], count = 4): Service[] {

   428→function parseJson<T>(raw: string | null | undefined, fallback: T): T {
   429→  if (!raw) return fallback;
   430→  try {
   431→    const v = JSON.parse(raw);
   432→    return (v === null || v === undefined) ? fallback : (v as T);
   433→  } catch {
   434→    return fallback;
   435→  }
   436→}
   437→
   438→function str(v: unknown, fallback = ""): string {
   439→  return typeof v === "string" ? v : fallback;
   440→}
   441→
   442→function ostr(v: unknown): string | undefined {
   443→  const s = str(v).trim();
   444→  return s.length ? s : undefined;
   445→}
   446→
   447→function num(v: unknown, fallback: number): number {
   448→  const n = Number(v);
   449→  return Number.isFinite(n) ? n : fallback;
   450→}
   451→
   452→function bool(v: unknown): boolean | undefined {
   453→  return typeof v === "boolean" ? v : undefined;
   454→}

   667→export type HeroSlideRow = {
   668→  id: string; scope: string; sort: number; kind: "image" | "youtube";
   669→  src: string; alt?: string; active: boolean;
   670→};
   671→
   672→export function mapHeroSlide(row: Record<string, unknown>): HeroSlideRow {
   673→  const src = str(row.src);
   674→  let kind: "image" | "youtube" = str(row.kind, "image") === "youtube" ? "youtube" : "image";
   675→  if (kind === "image" && parseYouTubeId(src)) kind = "youtube";
   676→  return {
   677→    id: str(row.id),
   678→    scope: str(row.scope, "home"),
   679→    sort: num(row.sort, 0),
   680→    kind,
   681→    src,
   682→    alt: ostr(row.alt),
   683→    active: row.active === undefined ? true : Boolean(row.active),
   684→  };
   685→}
   686→
   687→export function heroSlideRowToSlide(row: HeroSlideRow): HeroSlide {
   688→  if (row.kind === "youtube") {
   689→    const id = parseYouTubeId(row.src);
   690→    if (id) return { type: "youtube", id };
   691→    return { type: "image", src: row.src, alt: row.alt ?? "" };
   692→  }
   693→  return { type: "image", src: row.src, alt: row.alt ?? "" };
   694→}
   695→
   696→export function buildHomeHeroSlides(rows: HeroSlideRow[], videoUrls: string[]): HeroSlide[] {
   697→  const images = rows
   698→    .filter((r) => r.active && r.kind === "image")
   699→    .sort((a, b) => a.sort - b.sort)
   700→    .map(heroSlideRowToSlide);
   701→  const dbVideos = rows
   702→    .filter((r) => r.active && r.kind === "youtube")
   703→    .sort((a, b) => a.sort - b.sort)
   704→    .map(heroSlideRowToSlide)
   705→    .filter((s) => s.type === "youtube");
   706→  const settingsVideos = videoUrls
   707→    .map(parseYouTubeId)
   708→    .filter((id): id is string => Boolean(id))
   709→    .map((id) => ({ type: "youtube", id }) as HeroSlide);
   710→  const slides = buildHeroSlides(images, [...dbVideos, ...settingsVideos]);
   711→  return slides.length ? slides : buildHeroSlides(homeHeroImages, settingsVideos);
   712→}
