    80→    80→  if (!iso) return "";

    82→    82→    lang === "bn" ? "bn-BD" : "en-GB",
    83→    83→    {
    84→    84→      weekday: "long",
    85→    85→      day: "numeric",
    86→    86→      month: "long",
    87→    87→      year: "numeric",
    88→    88→    }
    89→    89→  );
    90→    90→}

    92→    92→function formatShortDate(iso: string, lang: Lang): string {

    96→    96→    year: "numeric",
