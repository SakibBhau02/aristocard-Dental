    93→function toIsoDate(d: Date): string {
    94→  const p = (n: number) => String(n).padStart(2, "0");
    95→  return `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())}`;
    96→}
    97→function localToday(): string {
    98→  return toIsoDate(new Date());
    99→}
   100→function addDaysIso(iso: string, days: number): string {
   101→  const [y, m, d] = iso.split("-").map(Number);
   102→  return toIsoDate(new Date(y, m - 1, d + days));
   103→}

   214→function AdminGate({ onUnlock }: { onUnlock: () => void }) {
   215→  const [value, setValue] = useState("");
   216→  const [error, setError] = useState<string | null>(null);
   217→  const [checking, setChecking] = useState(false);
   218→
   219→  async function handleSubmit(e: React.FormEvent) {
   220→    e.preventDefault();
   221→    if (checking) return;
   222→    setError(null);
   223→    setChecking(true);
   224→    try {
   225→      /* The passcode travels once, in this request body — the server
   226→         responds with an httpOnly session cookie that does the rest. */
   227→      const res = await fetch("/api/admin/login", {
   228→        method: "POST",
   229→        headers: { "Content-Type": "application/json" },
   230→        body: JSON.stringify({ code: value.trim() }),
   231→      });
   232→      if (res.status === 401) {
   233→        setError("Incorrect passcode. Please try again.");
   234→        return;
   235→      }
   236→      if (res.status === 429) {
   237→        setError("Too many attempts. Please wait a few minutes and try again.");
   238→        return;
   239→      }
   240→      if (!res.ok) throw new Error("Server error");
   241→      onUnlock();
   242→    } catch {
   243→      setError("Could not reach the server. Please try again.");
   244→    } finally {
   245→      setChecking(false);
   246→    }
   247→  }
   248→
   249→  return (
   250→    <div className="hero-mesh flex flex-1 items-center justify-center px-4 py-24">
   251→      <motion.div
   252→        initial={{ opacity: 0, y: 24 }}

   255→        className="w-full max-w-md"
   256→      >
   257→        <form
   258→          onSubmit={handleSubmit}
   259→          className="card-shine rounded-[2rem] border border-blue-100 bg-white p-8 shadow-[0_24px_60px_-24px_rgba(0,32,96,0.35)] sm:p-10"
   260→        >
   261→          <div className="flex items-center justify-center">
   262→            <span className="inline-flex h-16 w-16 items-center justify-center rounded-3xl bg-brand-navy text-white shadow-lg shadow-navy-900/30">
   263→              <Lock className="h-7 w-7" aria-hidden="true" />
   264→            </span>
   265→          </div>
   266→          <h1 className="font-heading mt-6 text-center text-2xl font-bold text-brand-navy">
   267→            Staff Access Only
   268→          </h1>
   269→          <p className="mt-2 text-center text-sm leading-relaxed text-slate-500">
   270→            Enter the clinic admin passcode to manage appointments. This area is
   271→            for Aristocrat Dental Care staff.
   272→          </p>
   273→
   274→          <label htmlFor="admin-code" className="sr-only">
   275→            Admin passcode
   276→          </label>
   277→          <input
   278→            id="admin-code"
   279→            type="password"
   280→            autoComplete="current-password"
   281→            value={value}
   282→            onChange={(e) => setValue(e.target.value)}
   283→            placeholder="Admin passcode"
   284→            className="mt-7 h-13 w-full rounded-2xl border border-blue-100 bg-ice px-5 text-sm text-brand-navy placeholder:text-slate-400 focus:border-brand-blue focus:outline-none focus:ring-2 focus:ring-brand-blue/25"
   285→          />

   287→          {error && (
   288→            <p className="mt-3 rounded-xl bg-rose-50 px-4 py-2.5 text-xs font-medium text-rose-600 ring-1 ring-rose-100">
   289→              {error}
   290→            </p>
   291→          )}
   292→
   293→          <Button
   294→            type="submit"
   295→            disabled={checking || value.trim().length === 0}
   296→            className="mt-5 h-12 w-full rounded-2xl bg-brand-navy text-sm font-bold text-white shadow-lg shadow-navy-900/25 hover:bg-brand-blue disabled:opacity-60"
   297→          >
   298→            {checking ? (
   299→              <Loader2 className="h-4.5 w-4.5 animate-spin" aria-hidden="true" />
   300→            ) : (
   301→              <ShieldCheck className="h-4.5 w-4.5" aria-hidden="true" />
   302→            )}
   303→            Unlock Dashboard
   304→          </Button>
   305→
   306→          <p className="mt-5 text-center text-[11px] leading-relaxed text-slate-400">
   307→            You'll stay signed in for 8 hours on this device. Demo passcode:{" "}
   308→            <code className="rounded bg-slate-100 px-1.5 py-0.5 font-mono text-[10px] text-slate-600">
   309→              aristocrat-2026
   310→            </code>
   311→          </p>
   312→        </form>
   313→      </motion.div>
   314→    </div>
   315→  );

   461→function AppointmentsPanel({
   462→  onLogout,
   463→  onSessionExpired,
   464→}: {
   465→  onLogout: () => void;
   466→  onSessionExpired: () => void;
   467→}) {
   468→  const [rows, setRows] = useState<AppointmentRow[]>([]);
   469→  const [stats, setStats] = useState<Stats | null>(null);
   470→  const [loading, setLoading] = useState(true);
   471→  const [loadError, setLoadError] = useState<string | null>(null);
   472→  const [filter, setFilter] = useState<StatusFilter>("ALL");
   473→  const [search, setSearch] = useState("");
   474→  const [busyRef, setBusyRef] = useState<string | null>(null);
   475→  const [lastUpdated, setLastUpdated] = useState<Date | null>(null);
   476→  const [preset, setPreset] = useState<DatePreset>("all");
   477→  const [fromDate, setFromDate] = useState("");
   478→  const [toDate, setToDate] = useState("");
   479→  const [range, setRange] = useState<{
   480→    from: string | null;
   481→    to: string | null;
   482→    label: string;
   483→    total: number;
   484→  } | null>(null);
   485→  const [exporting, setExporting] = useState(false);
   486→  const loadIdRef = useRef(0);
   487→
   488→  const load = useCallback(async () => {
   489→    const requestId = ++loadIdRef.current;
   490→    setLoading(true);
   491→    setLoadError(null);
   492→    try {
   493→      const qs = new URLSearchParams();
   494→      if (fromDate) qs.set("from", fromDate);
   495→      if (toDate) qs.set("to", toDate);
   496→      const query = qs.toString();
   497→      const res = await fetch(
   498→        `/api/admin/appointments${query ? `?${query}` : ""}`
   499→      );
   500→      if (res.status === 401) {
   501→        onSessionExpired();
   502→        return;
   503→      }
   504→      const data = await res.json();
   505→      if (!res.ok) throw new Error(data.error || "Failed to load");
   506→      if (requestId !== loadIdRef.current) return; /* a newer request is in flight */
   507→      setRows(data.appointments ?? []);
   508→      setStats(data.stats ?? null);
   509→      setRange(data.range ?? null);
   510→      setLastUpdated(new Date());
   511→    } catch (err) {
   512→      if (requestId !== loadIdRef.current) return;
   513→      setLoadError(err instanceof Error ? err.message : "Failed to load appointments.");
   514→    } finally {
   515→      if (requestId === loadIdRef.current) setLoading(false);
   516→    }
   517→  }, [fromDate, toDate, onSessionExpired]);
   518→
   519→  useEffect(() => {
   520→    load();
   521→  }, [load]);
   522→
   523→  /* auto-refresh every 60s (pause when tab hidden) */
   524→  useEffect(() => {
   525→    const id = window.setInterval(() => {
   526→      if (document.visibilityState === "visible") load();
   527→    }, 60_000);
   528→    return () => window.clearInterval(id);
   529→  }, [load]);
   530→
   531→  async function updateStatus(refCode: string, status: string) {
   532→    setBusyRef(refCode);
   533→    try {
   534→      const res = await fetch("/api/admin/appointments", {
   535→        method: "POST",
   536→        headers: {
   537→          "Content-Type": "application/json",
   538→        },
   539→        body: JSON.stringify({ refCode, status }),
   540→      });
   541→      const data = await res.json();
   542→      if (res.status === 401) {
   543→        onSessionExpired();
   544→        return;
   545→      }
   546→      if (!res.ok) throw new Error(data.error || "Update failed");
   547→      setRows((prev) =>
   548→        prev.map((r) => (r.refCode === refCode ? { ...r, status } : r))
   549→      );
   550→      setStats((prev) => {
   551→        if (!prev) return prev;
   552→        const statusOf = (r: AppointmentRow) =>
   553→          r.refCode === refCode ? status : r.status;
   554→        const today = localToday();
   555→        return {
   556→          ...prev,
   557→          pending: rows.filter((r) => statusOf(r) === "PENDING").length,
   558→          confirmed: rows.filter((r) => statusOf(r) === "CONFIRMED").length,
   559→          completed: rows.filter((r) => statusOf(r) === "COMPLETED").length,
   560→          cancelled: rows.filter((r) => statusOf(r) === "CANCELLED").length,
   561→          today: rows.filter((r) => r.date === today && statusOf(r) !== "CANCELLED").length,
   562→          upcoming: rows.filter(
   563→            (r) => r.date >= today && (statusOf(r) === "PENDING" || statusOf(r) === "CONFIRMED")
   564→          ).length,
   565→        };
   566→      });
   567→    } catch {
   568→      setLoadError("Could not update status. Please retry.");
   569→    } finally {
   570→      setBusyRef(null);
   571→    }
   572→  }
