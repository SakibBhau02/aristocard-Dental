   380→function ReadingProgress() {
   381→  const [progress, setProgress] = useState(0);
   382→
   383→  useEffect(() => {
   384→    let raf = 0;
   385→    const update = () => {
   386→      raf = 0;
   387→      const el = document.documentElement;
   388→      const total = el.scrollHeight - el.clientHeight;
   389→      setProgress(total > 0 ? Math.min(1, el.scrollTop / total) : 0);
   390→    };
   391→    const onScroll = () => {
   392→      if (!raf) raf = requestAnimationFrame(update);
   393→    };
   394→    update();
   395→    window.addEventListener("scroll", onScroll, { passive: true });
   396→    window.addEventListener("resize", onScroll, { passive: true });
   397→    return () => {
   398→      window.removeEventListener("scroll", onScroll);
   399→      window.removeEventListener("resize", onScroll);
   400→      if (raf) cancelAnimationFrame(raf);
   401→    };
   402→  }, []);
   403→
   404→  return (
   405→    <div
   406→      className="reading-progress"
   407→      style={{ transform: `scaleX(${progress})` }}
   408→      aria-hidden="true"
   409→    />
   410→  );
   411→}
