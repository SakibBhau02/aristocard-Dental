    73→function slugify(input: string): string {
    74→  return input
    75→    .toLowerCase()
    76→    .trim()
    77→    .replace(/[^a-z0-9\s-]/g, "")
    78→    .replace(/[\s_]+/g, "-")
    79→    .replace(/-+/g, "-")
    80→    .replace(/^-|-$/g, "")
    81→    .slice(0, 80);
    82→}

   154→function ChipsEditor({
   155→  label,
   156→  hint,
   157→  value,
   158→  onChange,
   159→  placeholder,
   160→  addLabel = "Add",
   161→}: {
   162→  label: string;
   163→  hint?: string;
   164→  value: string[];
   165→  onChange: (next: string[]) => void;
   166→  placeholder?: string;
   167→  addLabel?: string;
   168→}) {
   169→  const [pending, setPending] = useState("");
   170→
   171→  function add() {
   172→    const v = pending.trim();
   173→    if (!v) return;
   174→    if (!value.includes(v)) onChange([...value, v]);
   175→    setPending("");
   176→  }
   177→
   178→  return (
   179→    <div className="min-w-0">
   180→      <p className="text-xs font-bold uppercase tracking-wider text-slate-500">{label}</p>
   181→      {hint ? <p className="mt-0.5 text-xs leading-relaxed text-slate-400">{hint}</p> : null}
   182→      {value.length ? (
   183→        <div className="mt-2 flex flex-wrap gap-1.5">
   184→          {value.map((chip, i) => (
   185→            <span
   186→              key={`${chip}-${i}`}
   187→              className="inline-flex items-center gap-1.5 rounded-full bg-blue-50 py-1 pl-3 pr-1.5 text-xs font-semibold text-brand-navy ring-1 ring-blue-100"
   188→            >
   189→              {chip}
   190→              <button
   191→                type="button"
   192→                onClick={() => onChange(value.filter((_, j) => j !== i))}
   193→                aria-label={`Remove ${chip}`}
   194→                title={`Remove ${chip}`}
   195→                className="inline-flex h-5 w-5 items-center justify-center rounded-full text-slate-400 transition hover:bg-rose-50 hover:text-rose-600"
   196→              >
   197→                <X className="h-3 w-3" aria-hidden="true" />
   198→              </button>
   199→            </span>
   200→          ))}
   201→        </div>
   202→      ) : null}
   203→      <div className="mt-2 flex gap-2">
   204→        <Input
   205→          value={pending}
   206→          onChange={(e) => setPending(e.target.value)}
   207→          onKeyDown={(e) => {
   208→            if (e.key === "Enter") {
   209→              e.preventDefault();
   210→              add();
   211→            }
   212→          }}
   213→          placeholder={placeholder ?? "Type and press Enter"}
   214→          aria-label={`${label} — add item`}
   215→          className="h-9 rounded-xl border-blue-100 bg-ice text-sm text-brand-navy placeholder:text-slate-400 focus-visible:border-brand-blue focus-visible:ring-brand-blue/25"
   216→        />
   217→        <Button
   218→          type="button"
   219→          variant="outline"
   220→          size="sm"
   221→          onClick={add}
   222→          className="shrink-0 rounded-full border-blue-200 text-brand-navy hover:bg-blue-50"
   223→        >
   224→          <Plus className="h-4 w-4" aria-hidden="true" />
   225→          {addLabel}
   226→        </Button>
   227→      </div>
   228→    </div>
   229→  );
   230→}

   327→function DoctorsEditor({
   328→  row,
   329→  nextSort,
   330→  onSessionExpired,
   331→  onSaved,
   332→  onClose,
   333→}: {
   334→  row: DoctorRow | null;
   335→  nextSort: number;
   336→  onSessionExpired: () => void;
   337→  onSaved: (row: DoctorRow) => void;
   338→  onClose: () => void;
   339→}) {
   340→  const { post } = useAdminFetch(onSessionExpired);
   341→  const { ok: toastOk, fail: toastFail } = useAdminToast();
   342→  const { refresh } = useContent();
   343→  const [draft, setDraft] = useState<DoctorDraft>(() => rowToDraft(row));
   344→  const [slugEdited, setSlugEdited] = useState(row !== null);
   345→  const [formError, setFormError] = useState<string | null>(null);
   346→  const [saving, setSaving] = useState(false);
   347→  const [snapshot] = useState(() => JSON.stringify(rowToDraft(row)));
   348→
   349→  const dirty = JSON.stringify(draft) !== snapshot;
   350→
   351→  function update(patch: Partial<DoctorDraft>) {
   352→    setFormError(null);
   353→    setDraft((d) => ({ ...d, ...patch }));
   354→  }
   355→
   356→  function setName(value: string) {
   357→    setFormError(null);
   358→    setDraft((d) => ({
   359→      ...d,
   360→      name: value,
   361→      slug: slugEdited ? d.slug : slugify(value),
   362→    }));
   363→  }
   364→
   365→  function toggleDay(day: string) {
   366→    setFormError(null);
   367→    setDraft((d) => ({
   368→      ...d,
   369→      consultationDays: d.consultationDays.includes(day)
   370→        ? d.consultationDays.filter((x) => x !== day)
   371→        : [...d.consultationDays, day],
   372→    }));
   373→  }
   374→
   375→  function back() {
   376→    if (dirty && !window.confirm("Discard unsaved changes to this doctor?")) return;
   377→    onClose();
   378→  }
   379→
   380→  function reset() {
   381→    setDraft(JSON.parse(snapshot) as DoctorDraft);
   382→    setSlugEdited(row !== null);
   383→    setFormError(null);
   384→  }
   385→
   386→  async function save() {
   387→    if (saving) return;
   388→    const d = draft;
   389→    const problems: string[] = [];
   390→
   391→    const name = d.name.trim();
   392→    if (name.length < 3) problems.push("Doctor name must be at least 3 characters.");
   393→    const slug = d.slug.trim().toLowerCase();
   394→    if (!SLUG_RE.test(slug) || slug.length < 2 || slug.length > 80)
   395→      problems.push("Slug must be lowercase letters, numbers and hyphens (2–80 characters).");
   396→    if (d.role.trim().length < 2) problems.push("Role is required.");
   397→    if (d.department.trim().length < 2) problems.push("Department is required.");
   398→    if (d.bio.trim().length < 10) problems.push("Bio must be at least 10 characters.");
   399→    const years = Number(d.experienceYears);
   400→    const patients = Number(d.patientsTreated);
   401→    const rating = Number(d.rating);
   402→    if (!Number.isFinite(years) || years < 0 || years > 80)
   403→      problems.push("Experience years must be a number between 0 and 80.");
   404→    if (!Number.isFinite(patients) || patients < 0)
   405→      problems.push("Patients treated must be a positive number.");
   406→    if (!Number.isFinite(rating) || rating < 0 || rating > 5)
   407→      problems.push("Rating must be a number between 0 and 5.");
   408→
   409→    if (problems.length) {
   410→      setFormError(problems[0]);
   411→      toastFail("Check the form", problems[0]);
   412→      return;
   413→    }
   414→
   415→    setSaving(true);
   416→    const res = await post<{ doctor?: DoctorRow; message?: string; error?: string }>(
   417→      "/api/admin/doctors",
   418→      {
   419→        ...(row ? { id: row.id, sort: row.sort } : { sort: nextSort }),
   420→        name,
   421→        slug,
   422→        role: d.role.trim(),
   423→        roleBn: d.roleBn.trim() || undefined,
   424→        department: d.department.trim(),
   425→        departmentBn: d.departmentBn.trim() || undefined,
   426→        credentials: d.credentials,
   427→        bio: d.bio.trim(),
   428→        experienceYears: years,
   429→        patientsTreated: patients,
   430→        rating,
   431→        specializations: d.specializations,
   432→        languages: d.languages,
   433→        consultationDays: d.consultationDays,
   434→        accent: d.accent,
   435→        photo: d.photo.trim() || undefined,
   436→        active: d.active,
   437→      }
   438→    );
   439→    setSaving(false);
   440→
   441→    if (res.ok && res.data?.doctor) {
   442→      toastOk(row ? "Doctor updated" : "Doctor created", res.data.message);
   443→      void refresh();
   444→      onSaved(res.data.doctor);
   445→      return;
   446→    }
   447→    const message =
   448→      (res.data as { error?: string } | null)?.error ??
   449→      "Could not save this doctor. Please try again.";
   450→    setFormError(message);
   451→    toastFail("Save failed", message);
   452→  }
   453→
   454→  const accentSafe = /^#[0-9a-fA-F]{6}$/.test(draft.accent) ? draft.accent : "#012c5f";
   455→
   456→  return (
   457→    <div className="space-y-5">
   458→      {/* Editor top bar */}
   459→      <div className="flex flex-wrap items-center gap-3">
   460→        <Button
   461→          type="button"
   462→          variant="outline"
   463→          size="sm"
   464→          onClick={back}
   465→          className="rounded-full border-blue-200 text-slate-600 hover:bg-blue-50 hover:text-brand-navy"
   466→        >
   467→          <ArrowLeft className="h-4 w-4" aria-hidden="true" />
   468→          Back to list
   469→        </Button>
   470→        <div className="min-w-0">
   471→          <h2 className="font-heading text-base font-bold text-brand-navy">
   472→            {row ? "Edit doctor" : "Add a new doctor"}
   473→          </h2>
   474→          <p className="truncate text-xs text-slate-500">
   475→            {row
   476→              ? `${row.name} · public profile /doctors/${row.slug}`
   477→              : "New doctors appear on the public roster as soon as you save."}
   478→          </p>
   479→        </div>
   480→        {row ? (
   481→          <span className="ml-auto">
   482→            {row.active ? (
   483→              <StatusPill tone="green">
   484→                <BadgeCheck className="h-3 w-3" aria-hidden="true" />
   485→                Active
   486→              </StatusPill>
   487→            ) : (
   488→              <StatusPill tone="slate">
   489→                <EyeOff className="h-3 w-3" aria-hidden="true" />
   490→                Hidden
   491→              </StatusPill>
   492→            )}
   493→          </span>
   494→        ) : null}
   495→      </div>
   496→
   497→      {/* Identity */}
   498→      <EditorCard title="Identity" description="Name, public slug, accent colour and profile photo.">
   499→        <div className="grid gap-4 sm:grid-cols-2">
   500→          <Field label="Full name">
   501→            <Input
   502→              value={draft.name}
   503→              onChange={(e) => setName(e.target.value)}
   504→              placeholder="Dr. Full Name"
   505→              className={inputCls}
   506→            />
   507→          </Field>
   508→          <Field
   509→            label="Slug"
   510→            hint={
   511→              row
   512→                ? "⚠ Editing the slug changes the doctor's public profile link."
   513→                : "Auto-generated from the name — you can override it."
   514→            }
   515→          >
   516→            <Input
   517→              value={draft.slug}
   518→              onChange={(e) => {
   519→                setSlugEdited(true);
   520→                update({ slug: e.target.value });
   521→              }}
   522→              placeholder="dr-full-name"
   523→              className={cn(inputCls, "font-mono")}
   524→            />
   525→          </Field>
   526→        </div>
   527→        <div className="grid gap-4 sm:grid-cols-[auto_1fr] sm:items-start">
   528→          <Field label="Accent colour" hint="Used for card rings and detail accents.">
   529→            <div className="flex items-center gap-2">
   530→              <input
   531→                type="color"
   532→                value={accentSafe}
   533→                onChange={(e) => update({ accent: e.target.value })}
   534→                aria-label="Pick accent colour"
   535→                title="Pick accent colour"
   536→                className="h-10 w-14 cursor-pointer rounded-xl border border-blue-100 bg-white p-1"
   537→              />
   538→              <Input
   539→                value={draft.accent}
   540→                onChange={(e) => update({ accent: e.target.value })}
   541→                aria-label="Accent colour hex value"
   542→                className={cn(inputCls, "w-32 font-mono")}
   543→              />
   544→            </div>
   545→          </Field>
   546→          <ImageUploadField
   547→            label="Photo"
   548→            hint="Square portraits work best (min 400×400)."
   549→            value={draft.photo || undefined}
   550→            onChange={(url) => update({ photo: url ?? "" })}
   551→            onUnauthorized={onSessionExpired}
   552→          />
   553→        </div>
   554→        <SwitchRow
   555→          label="Visible on the website"
   556→          description="Hidden doctors stay in the database but disappear from the public roster."
   557→          checked={draft.active}
   558→          onCheckedChange={(checked) => update({ active: checked })}
   559→        />
   560→      </EditorCard>
   561→
   562→      {/* Role & bio */}
   563→      <EditorCard title="Role & bio" description="Shown on the doctors page and profile cards.">
   564→        <BilingualField
   565→          label="Role / title"
   566→          enValue={draft.role}
   567→          bnValue={draft.roleBn}
   568→          onEnChange={(v) => update({ role: v })}
   569→          onBnChange={(v) => update({ roleBn: v })}
   570→          placeholder="Chief Dental Surgeon & Founder"
   571→        />
   572→        <BilingualField
   573→          label="Department"
   574→          enValue={draft.department}
   575→          bnValue={draft.departmentBn}
   576→          onEnChange={(v) => update({ department: v })}
   577→          onBnChange={(v) => update({ departmentBn: v })}
   578→          placeholder="Advanced Dentistry & Orthodontics"
   579→        />
   580→        <Field label="Bio (English)" hint="2–3 sentences for the profile page.">
   581→          <Textarea
   582→            value={draft.bio}
   583→            onChange={(e) => update({ bio: e.target.value })}
   584→            rows={6}
   585→            placeholder="A short professional biography…"
   586→            className="border-blue-100 bg-ice text-sm text-brand-navy placeholder:text-slate-400 focus-visible:border-brand-blue focus-visible:ring-brand-blue/25"
   587→          />
   588→        </Field>
   589→      </EditorCard>
   590→
   591→      {/* Numbers */}
   592→      <EditorCard title="Experience numbers" description="Displayed as highlights on the doctor card.">
   593→        <div className="grid gap-4 sm:grid-cols-3">
   594→          <Field label="Years of experience">
   595→            <Input
   596→              type="number"
   597→              min={0}
   598→              max={80}

   709→  const [loaded, setLoaded] = useState(false);
   710→  const [loadBusy, setLoadBusy] = useState(false);

   712→  const [search, setSearch] = useState("");

   717→  const load = useCallback(async () => {
   718→    setLoadBusy(true);
   719→    try {
   720→      const res = await fetch("/api/admin/doctors");
   721→      if (res.status === 401) {
   722→        onSessionExpired();
   723→        return;
   724→      }
   725→      const data = (await res.json().catch(() => null)) as
   726→        | { doctors?: DoctorRow[]; error?: string }
   727→        | null;
   728→      if (!res.ok) throw new Error(data?.error ?? `Request failed (${res.status}).`);
   729→      setRows(data?.doctors ?? []);
   730→      setLoaded(true);
   731→      setLoadError(null);
   732→    } catch (err) {
   733→      setLoadError(err instanceof Error ? err.message : "Could not load doctors.");
   734→    } finally {
   735→      setLoadBusy(false);
   736→    }

   738→

   740→    void load();
