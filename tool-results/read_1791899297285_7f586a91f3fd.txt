   262→function toIsoDate(d: Date): string {
   263→  const p = (n: number) => String(n).padStart(2, "0");
   264→  return `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())}`;
   265→}
   266→
   267→export function localToday(): string {
   268→  return toIsoDate(new Date());

   271→function addDaysIso(iso: string, days: number): string {
   272→  const [y, m, d] = iso.split("-").map(Number);
   273→  return toIsoDate(new Date(y, m - 1, d + days));
   274→}

   295→/** "Oct 12, 2026" from an ISO date or full timestamp. */
   296→export function prettyDay(iso: string): string {
   297→  const d = new Date(iso);
   298→  if (Number.isNaN(d.getTime())) return iso;
   299→  return d.toLocaleDateString("en-US", {
   300→    month: "short",
   301→    day: "numeric",
   302→    year: "numeric",
   303→  });
   304→}
   305→
   306→/** "Oct 12, 2026 · 4:26 PM" from an ISO timestamp. */
   307→export function prettyStamp(iso: string): string {
   308→  const d = new Date(iso);
   309→  if (Number.isNaN(d.getTime())) return iso;
   310→  return `${prettyDay(iso)} · ${d.toLocaleTimeString("en-US", {
   311→    hour: "numeric",
   312→    minute: "2-digit",
   313→  })}`;
   314→}

   316→/** "3 days ago" style relative label (best-effort, local clock). */
   317→export function relativeTime(iso: string): string {
   318→  const then = new Date(iso).getTime();
   319→  if (Number.isNaN(then)) return "";
   320→  const mins = Math.round((Date.now() - then) / 60_000);
   321→  if (mins < 1) return "just now";
   322→  if (mins < 60) return `${mins} min ago`;
   323→  const hours = Math.round(mins / 60);
   324→  if (hours < 24) return `${hours} hr ago`;
   325→  const days = Math.round(hours / 24);
   326→  if (days < 7) return `${days} day${days === 1 ? "" : "s"} ago`;
   327→  const weeks = Math.round(days / 7);
   328→  if (weeks < 5) return `${weeks} wk ago`;
   329→  const months = Math.round(days / 30);
   330→  if (months < 12) return `${months} mo ago`;
   331→  return `${Math.round(months / 12)} yr ago`;
   332→}

   334→/** "Mon 29" chart label from a YYYY-MM-DD date. */
   335→export function chartDayLabel(iso: string): string {
   336→  const [y, m, d] = iso.split("-").map(Number);
   337→  if (!y || !m || !d) return iso;
   338→  const dt = new Date(Date.UTC(y, m - 1, d));
   339→  const weekday = dt.toLocaleDateString("en-US", {
   340→    timeZone: "UTC",
   341→    weekday: "short",
   342→  });
   343→  return `${weekday} ${d}`;
   344→}

   347→export function initialsOf(name: string): string {
   348→  const parts = name.trim().split(/\s+/).filter(Boolean);
   349→  if (parts.length === 0) return "?";
   350→  return parts

   352→    .map((p) => p.charAt(0).toUpperCase())
   353→    .join("");

   376→/** Shared CSV download — the filename comes from Content-Disposition. */
   377→export async function downloadCsvFile(
   378→  path: string,
   379→  fallbackName: string
   380→): Promise<"ok" | "unauthorized" | "error"> {
   381→  try {
   382→    const res = await fetch(path);
   383→    if (res.status === 401) return "unauthorized";
   384→    if (!res.ok) return "error";
   385→    const blob = await res.blob();
   386→    const disposition = res.headers.get("Content-Disposition") ?? "";
   387→    const filename =
   388→      /filename="([^"]+)"/.exec(disposition)?.[1] ?? fallbackName;
   389→    const url = URL.createObjectURL(blob);
   390→    const a = document.createElement("a");
   391→    a.href = url;
   392→    a.download = filename;
   393→    document.body.appendChild(a);
   394→    a.click();
   395→    a.remove();
   396→    URL.revokeObjectURL(url);
   397→    return "ok";
   398→  } catch {
   399→    return "error";
   400→  }
   401→}

   498→/* Panels fire several parallel requests; collapse 403 notices so a
   499→   mid-session role change doesn't spam the toaster. */
   500→let lastForbiddenNoticeAt = 0;
   501→
   502→function notifyForbidden(
   503→  fail: (title: string, description?: string) => void,
   504→  message: string
   505→) {
   506→  const now = Date.now();
   507→  if (now - lastForbiddenNoticeAt < 15_000) return;
   508→  lastForbiddenNoticeAt = now;
   509→  fail("Not allowed", message);
   510→}

   512→export function useAdminFetch(onUnauthorized: () => void) {
   513→  const [loading, setLoading] = useState(false);
   514→  const [error, setError] = useState<string | null>(null);
   515→  const onAuthRef = useRef(onUnauthorized);
   516→  onAuthRef.current = onUnauthorized;
   517→  const { fail: toastFail } = useAdminToast();
   518→  const failRef = useRef(toastFail);
   519→  failRef.current = toastFail;
   520→
   521→  const handle401 = useCallback((res: Response): boolean => {
   522→    if (res.status === 401) {
   523→      onAuthRef.current();
   524→      return true;
   525→    }
   526→    return false;
   527→  }, []);
   528→
   529→  const handle403 = useCallback((res: Response, message: string) => {
   530→    if (res.status === 403) {
   531→      notifyForbidden(failRef.current, message);
   532→      return true;
   533→    }
   534→    return false;
   535→  }, []);
   536→
   537→  const get = useCallback(
   538→    async <T,>(path: string): Promise<T | null> => {
   539→      setLoading(true);
   540→      setError(null);
   541→      try {
   542→        const res = await fetch(path);
   543→        if (handle401(res)) return null;
   544→        const data = await res.json().catch(() => null);
   545→        if (!res.ok) {
   546→          const message =
   547→            (data as { error?: string } | null)?.error ?? `Request failed (${res.status}).`;
   548→          handle403(res, message);
   549→          throw new Error(message);
   550→        }
   551→        return data as T;
   552→      } catch (err) {
   553→        setError(err instanceof Error ? err.message : "Request failed.");
   554→        return null;
   555→      } finally {
   556→        setLoading(false);
   557→      }
   558→    },
   559→    [handle401, handle403]
   560→  );
   561→
   562→  const post = useCallback(
   563→    async <T,>(
   564→      path: string,
   565→      payload: unknown
   566→    ): Promise<{ ok: boolean; data: T | null; status: number }> => {
   567→      setLoading(true);
   568→      setError(null);
   569→      try {
   570→        const res = await fetch(path, {
   571→          method: "POST",
   572→          headers: { "Content-Type": "application/json" },
   573→          body: JSON.stringify(payload ?? {}),
   574→        });
   575→        if (handle401(res)) return { ok: false, data: null, status: 401 };
   576→        const data = await res.json().catch(() => null);
   577→        if (!res.ok) {
   578→          const message =
   579→            (data as { error?: string } | null)?.error ?? `Request failed (${res.status}).`;
   580→          handle403(res, message);
   581→          setError(message);
   582→          return { ok: false, data: data as T | null, status: res.status };
   583→        }
   584→        return { ok: true, data: data as T, status: res.status };
   585→      } catch (err) {
   586→        setError(err instanceof Error ? err.message : "Request failed.");
   587→        return { ok: false, data: null, status: 0 };
   588→      } finally {
   589→        setLoading(false);
   590→      }
   591→    },
   592→    [handle401, handle403]
   593→  );
   594→
   595→  const del = useCallback(
   596→    async <T,>(path: string): Promise<{ ok: boolean; data: T | null; status: number }> => {
   597→      setLoading(true);
   598→      setError(null);
   599→      try {
   600→        const res = await fetch(path, { method: "DELETE" });
   601→        if (handle401(res)) return { ok: false, data: null, status: 401 };
   602→        const data = await res.json().catch(() => null);
   603→        if (!res.ok) {
   604→          const message =
   605→            (data as { error?: string } | null)?.error ?? `Request failed (${res.status}).`;
   606→          handle403(res, message);
   607→          setError(message);
   608→          return { ok: false, data: data as T | null, status: res.status };
   609→        }
   610→        return { ok: true, data: data as T, status: res.status };
   611→      } catch (err) {
   612→        setError(err instanceof Error ? err.message : "Request failed.");
   613→        return { ok: false, data: null, status: 0 };
   614→      } finally {
   615→        setLoading(false);
   616→      }
   617→    },
   618→    [handle401, handle403]
   619→  );
   620→
   621→  const clearError = useCallback(() => setError(null), []);
   622→
   623→  return { get, post, del, loading, error, handle401, clearError };
   624→}

   833→export function ImageUploadField({
   834→  value,
   835→  onChange,
   836→  label,
   837→  hint,
   838→  onUnauthorized,
   839→}: {
   840→  value: string | undefined;

   842→  label?: string;
   843→  hint?: string;
   844→  onUnauthorized?: () => void;
   845→}) {
   846→  const inputRef = useRef<HTMLInputElement | null>(null);
   847→  const [uploading, setUploading] = useState(false);
   848→  const { fail } = useAdminToast();
   849→
   850→  async function uploadFile(file: File) {
   851→    if (uploading) return;
   852→    setUploading(true);
   853→    try {
   854→      const body = new FormData();

   856→      const res = await fetch("/api/admin/upload", { method: "POST", body });
