    65→const GA4_RE = /^G-[A-Z0-9]{6,12}$/;
    66→const GTM_RE = /^GTM-[A-Z0-9]{5,10}$/;
    67→const FB_PIXEL_RE = /^\d{14,17}$/;
    68→const TIKTOK_RE = /^[A-Z0-9]{15,25}$/;
    69→const TELEGRAM_TOKEN_RE = /^\d{6,12}:[A-Za-z0-9_-]{30,60}$/;
    70→const TELEGRAM_CHAT_RE = /^(-?\d{4,15}|@[A-Za-z0-9_]{4,32})$/;

    72→/** Client-side issues mirroring the server's merged-config checks. */
    73→function clientIssues(form: IntegrationsConfig, storedToken: string): string[] {
    74→  const issues: string[] = [];
    75→  if (form.ga4Id && !GA4_RE.test(form.ga4Id))
    76→    issues.push("Google Analytics ID should look like G-ABC123DEF4.");
    77→  if (form.gtmId && !GTM_RE.test(form.gtmId))
    78→    issues.push("Tag Manager container ID should look like GTM-ABC1234.");
    79→  if (form.fbPixelId && !FB_PIXEL_RE.test(form.fbPixelId))
    80→    issues.push("Facebook Pixel ID must be 14–17 digits.");
    81→  if (form.tiktokPixelId && !TIKTOK_RE.test(form.tiktokPixelId))
    82→    issues.push("TikTok Pixel ID must be 15–25 letters/digits.");
    83→  /* A masked token means "unchanged" — validate the stored secret. */
    84→  const token = form.telegramBotToken.includes("•") ? storedToken : form.telegramBotToken;
    85→  if (token && !TELEGRAM_TOKEN_RE.test(token))
    86→    issues.push("Telegram bot token should look like 123456789:AAE… (copy it from @BotFather).");
    87→  if (form.telegramChatId && !TELEGRAM_CHAT_RE.test(form.telegramChatId))
    88→    issues.push("Telegram chat ID should be numeric (or @channelname).");
    89→  if (form.telegramChatId && !token) issues.push("Add the bot token to use the chat ID.");
    90→  if (token && !form.telegramChatId)
    91→    issues.push("Add the chat ID to receive Telegram alerts.");
    92→  return issues;
    93→}

   371→export function IntegrationsPanel({ onSessionExpired }: PanelProps) {
   372→  const { get, post, error, clearError } = useAdminFetch(onSessionExpired);
   373→  const { ok, fail } = useAdminToast();
   374→
   375→  const [snapshot, setSnapshot] = useState<IntegrationsConfig | null>(null);
   376→  const [form, setForm] = useState<IntegrationsConfig | null>(null);
   377→  const [role, setRole] = useState<"OWNER" | "EDITOR" | "VIEWER">("OWNER");
   378→  const [loadFailed, setLoadFailed] = useState(false);
   379→  const [loadTick, setLoadTick] = useState(0);
   380→  const [saving, setSaving] = useState(false);
   381→  const [testing, setTesting] = useState(false);
   382→  const [showToken, setShowToken] = useState(false);
   383→  const [testNote, setTestNote] = useState("");
   384→  const [serverIssues, setServerIssues] = useState<string[]>([]);
   385→
   386→  /* Initial fetch (+ manual retry) — same shape as the settings panel. */
   387→  useEffect(() => {
   388→    let cancelled = false;
   389→    (async () => {
   390→      const data = await get<{ integrations: IntegrationsConfig; role?: string }>(
   391→        "/api/admin/integrations"
   392→      );
   393→      if (cancelled) return;
   394→      if (!data) {
   395→        setLoadFailed(true);
   396→        return;
   397→      }
   398→      setLoadFailed(false);
   399→      setSnapshot(clone(data.integrations));
   400→      setForm(clone(data.integrations));
   401→      setServerIssues([]);
   402→      setRole(
   403→        data.role === "EDITOR" || data.role === "VIEWER" ? data.role : "OWNER"
   404→      );
   405→    })();
   406→    return () => {
   407→      cancelled = true;
   408→    };
   409→  }, [get, loadTick]);
   410→
   411→  const dirty = useMemo(
   412→    () => (snapshot && form ? JSON.stringify(form) !== JSON.stringify(snapshot) : false),
   413→    [snapshot, form]
   414→  );
   415→
   416→  /* Non-OWNER reads are read-only (the nav hides this module for them;
   417→     this guards a mid-session role change). Server 403s mutations anyway. */
   418→  const readOnly = role !== "OWNER";
   419→  const telegramConnected = Boolean(snapshot?.telegramBotToken && snapshot?.telegramChatId);
   420→
   421→  const set = useCallback(<K extends keyof IntegrationsConfig>(
   422→    key: K,
   423→    value: IntegrationsConfig[K]
   424→  ) => {
   425→    setForm((f) => (f ? { ...f, [key]: value } : f));
   426→  }, []);
   427→
   428→  async function save() {
   429→    if (!form || !snapshot || saving) return;
   430→    /* Live client-side gate (mirrors the server regexes). */
   431→    const issues = clientIssues(form, snapshot.telegramBotToken);
   432→    if (issues.length > 0) {
   433→      setServerIssues([]);
   434→      fail("Check the highlighted fields", issues[0]);
   435→      return;
   436→    }
   437→    setSaving(true);
   438→    const payload: IntegrationsConfig & { clearTelegramToken?: boolean } = { ...form };
   439→    /* Clearing/emptying the token removes the bot on purpose. */
   440→    if (!form.telegramBotToken && snapshot.telegramBotToken) {
   441→      payload.clearTelegramToken = true;
   442→    }
   443→    const res = await post<{
   444→      integrations?: IntegrationsConfig;
   445→      message?: string;
   446→      error?: string;
   447→      issues?: string[];
   448→    }>("/api/admin/integrations", payload);
   449→    setSaving(false);
   450→    if (res.ok && res.data?.integrations) {
   451→      const saved = res.data.integrations;
   452→      setSnapshot(clone(saved));
   453→      setForm(clone(saved));
   454→      setServerIssues([]);
   455→      ok("Integrations saved", res.data.message ?? "Pixels and alerts are live on the public site.");
   456→    } else {
   457→      setServerIssues(
   458→        res.data?.issues ??
   459→          (res.data?.error ? [res.data.error] : ["Please check the fields and try again."])
   460→      );
   461→      fail("Could not save integrations", res.data?.error ?? "Please try again.");
   462→    }
   463→  }
   464→
   465→  function reset() {
   466→    if (snapshot) setForm(clone(snapshot));
   467→    setServerIssues([]);
   468→    setTestNote("");
   469→    clearError();
   470→  }
   471→
   472→  /* Test button: sends with the CURRENT UNSAVED form values. A masked
   473→     token is skipped so the server tests the stored secret instead. */
   474→  async function sendTest() {
   475→    if (!form || testing) return;
   476→    setTesting(true);
   477→    const payload: { message?: string; telegramBotToken?: string; telegramChatId?: string } = {};
   478→    const note = testNote.trim();
   479→    if (note) payload.message = note.slice(0, 300);
   480→    const token = form.telegramBotToken.trim();
   481→    if (!token.includes("•")) {
   482→      if (token) payload.telegramBotToken = token;
   483→      const chat = form.telegramChatId.trim();
   484→      if (chat) payload.telegramChatId = chat;
   485→    }
   486→    const res = await post<{ success?: boolean; message?: string; error?: string }>(
   487→      "/api/admin/integrations/test",
   488→      payload
   489→    );
   490→    setTesting(false);
   491→    if (res.ok) {
   492→      ok("Test message sent", res.data?.message ?? "Check Telegram on your phone.");
   493→    } else {
   494→      fail("Test failed", res.data?.error ?? "The test message could not be sent.");
   495→    }
   496→  }
   497→
   498→  const tokenInvalid =
   499→    form &&
   500→    form.telegramBotToken.trim().length > 0 &&
   501→    !form.telegramBotToken.includes("•") &&
   502→    !TELEGRAM_TOKEN_RE.test(form.telegramBotToken.trim());
   503→  const chatInvalid =
   504→    form && form.telegramChatId.trim().length > 0 && !TELEGRAM_CHAT_RE.test(form.telegramChatId.trim());
   505→
   506→  /* ---- render ---- */
   507→
   508→  if (!snapshot || !form) {
   509→    if (loadFailed) {
   510→      return (
   511→        <div className="space-y-4">
   512→          <PanelHeader title="Integrations & Tracking" />
   513→          <ErrorNote message={error ?? "Could not load the integration settings."} />
   514→          <div>
   515→            <Button
   516→              onClick={() => {
   517→                setLoadFailed(false);
   518→                clearError();
   519→                setLoadTick((t) => t + 1);
   520→              }}
   521→              variant="outline"
   522→              size="sm"
   523→              className="rounded-full border-blue-200 text-brand-navy hover:bg-blue-50"
   524→            >
   525→              <RefreshCw className="h-4 w-4" aria-hidden="true" />
   526→              Try again
   527→            </Button>
   528→          </div>
   529→        </div>
   530→      );
   531→    }
   532→    return <PanelSkeleton />;
   533→  }
   534→
   535→  return (
   536→    <div className="space-y-4">
   537→      <PanelHeader
   538→        title="Integrations & Tracking"
   539→        description="Connect marketing pixels, get Telegram alerts on your phone, and switch on privacy-safe first-party analytics."
   540→      />
   541→      {error ? <ErrorNote message={error} /> : null}
   542→      {serverIssues.length > 0 ? <IssuesNote issues={serverIssues} /> : null}
   543→      {readOnly ? (
