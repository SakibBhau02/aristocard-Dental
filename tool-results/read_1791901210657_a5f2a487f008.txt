   833→function ReadingProgress() {
   834→  const [progress, setProgress] = useState(0);
   835→
   836→  useEffect(() => {
   837→    let raf = 0;
   838→    const update = () => {
   839→      raf = 0;
   840→      const el = document.documentElement;
   841→      const total = el.scrollHeight - el.clientHeight;
   842→      setProgress(total > 0 ? Math.min(1, el.scrollTop / total) : 0);
   843→    };
   844→    const onScroll = () => {
   845→      if (!raf) raf = requestAnimationFrame(update);
   846→    };
   847→    update();
   848→    window.addEventListener("scroll", onScroll, { passive: true });
   849→    window.addEventListener("resize", onScroll, { passive: true });
   850→    return () => {
   851→      window.removeEventListener("scroll", onScroll);
   852→      window.removeEventListener("resize", onScroll);
   853→      if (raf) cancelAnimationFrame(raf);
   854→    };
   855→  }, []);
   856→
   857→  return (
   858→    <div
   859→      className="reading-progress"
   860→      style={{ transform: `scaleX(${progress})` }}
   861→      aria-hidden="true"
   862→    />
   863→  );
   864→}

   866→/* Active-section tracking for the article table of contents.
   867→   `routeKey` (the post slug) re-runs the observer when navigating
   868→   between articles that happen to have the same section count; the
   869→   active index is DERIVED against the key so it resets on navigation
   870→   without setState-in-effect. */
   871→function useActiveSection(count: number, routeKey: string) {
   872→  const [state, setState] = useState({ key: routeKey, active: 0 });
   873→  const active = state.key === routeKey ? state.active : 0;
   874→
   875→  useEffect(() => {
   876→    if (count < 2) return;
   877→    const headings = Array.from({ length: count }, (_, i) =>
   878→      document.getElementById(`section-${i}`)
   879→    );
   880→    const visible = headings.filter(Boolean) as HTMLElement[];
   881→    if (visible.length === 0) return;
   882→    const observer = new IntersectionObserver(
   883→      (entries) => {
   884→        entries.forEach((entry) => {
   885→          if (entry.isIntersecting) {
   886→            const idx = Number(entry.target.id.replace("section-", ""));
   887→            setState({ key: routeKey, active: idx });
   888→          }
   889→        });
   890→      },
   891→      { rootMargin: "-25% 0px -65% 0px", threshold: 0 }
   892→    );
   893→    visible.forEach((h) => observer.observe(h));
   894→    return () => observer.disconnect();
   895→  }, [count, routeKey]);
   896→  return [active, (i: number) => setState({ key: routeKey, active: i })] as const;
   897→}
