   302→/* Defensive healer for About blobs saved before a field existed.
   303→   deepMerge replaces ARRAYS wholesale, so an old blob's timeline items
   304→   keep lacking year/icon even after merging — refill per-index from
   305→   the defaults so the public bundle is always complete. */
   306→export function normalizeAboutContent(about: AboutContent): AboutContent {
   307→  const defaults = defaultAboutContent();
   308→  const obj = (v: unknown): Record<string, unknown> =>
   309→    v !== null && typeof v === "object" ? (v as Record<string, unknown>) : {};
   310→  const timeline = (Array.isArray(about.timeline) ? about.timeline : defaults.timeline).map(
   311→    (t, i) => {
   312→      const raw = obj(t);
   313→      return {
   314→        year: str(raw.year) || defaults.timeline[i]?.year || "2024",
   315→        icon: str(raw.icon) || defaults.timeline[i]?.icon || "Sparkles",
   316→        title: str(raw.title),
   317→        titleBn: ostr(raw.titleBn),
   318→        description: str(raw.description),
   319→        descriptionBn: ostr(raw.descriptionBn),
   320→      };
   321→    }
   322→  );
   323→  const storyStats = Array.isArray(about.storyStats)
   324→    ? about.storyStats.map((s) => {
   325→        const raw = obj(s);
   326→        return {
   327→          value: str(raw.value),
   328→          label: str(raw.label),
   329→          labelBn: ostr(raw.labelBn),
   330→        };
   331→      })
   332→    : defaults.storyStats;
   333→  return {
   334→    ...about,
   335→    timeline,
   336→    storyStats,
   337→    founderName: str(about.founderName).trim() || defaults.founderName,
   338→    founderImage:
   339→      about.founderImage === undefined ? defaults.founderImage : str(about.founderImage),
   340→    founderImageAlt:
   341→      about.founderImageAlt === undefined ? defaults.founderImageAlt : str(about.founderImageAlt),
   342→  };
   343→}

   406→/* ------------------------------------------------------------------ */
   407→/*  Deep merge — DB sparse overrides over defaults                     */
   408→/* ------------------------------------------------------------------ */
   409→
   410→export function deepMerge<T>(base: T, override: unknown): T {
   411→  if (override === null || override === undefined) return base;
   412→  if (Array.isArray(base) || Array.isArray(override)) {
   413→    /* Arrays are replaced wholesale (never merged item-by-item). */
   414→    return (Array.isArray(override) ? override : base) as T;
   415→  }
   416→  if (typeof base === "object" && typeof override === "object" && base !== null) {
   417→    const out: Record<string, unknown> = { ...(base as Record<string, unknown>) };
   418→    for (const [k, v] of Object.entries(override as Record<string, unknown>)) {
   419→      const b = (base as Record<string, unknown>)[k];
   420→      out[k] = b !== undefined && b !== null && typeof b === "object" && !Array.isArray(b) && v !== null && typeof v === "object" && !Array.isArray(v)
   421→        ? deepMerge(b, v)
   422→        : v;
   423→    }
   424→    return out as T;
   425→  }
   426→  return (override as T) ?? base;
   427→}

   429→export function featuredServices(list: Service[], count = 4): Service[] {

   438→function parseJson<T>(raw: string | null | undefined, fallback: T): T {
   439→  if (!raw) return fallback;
   440→  try {
   441→    const v = JSON.parse(raw);
   442→    return (v === null || v === undefined) ? fallback : (v as T);
   443→  } catch {
   444→    return fallback;
   445→  }
   446→}
   447→
   448→function str(v: unknown, fallback = ""): string {
   449→  return typeof v === "string" ? v : fallback;
   450→}
   451→
   452→function ostr(v: unknown): string | undefined {
   453→  const s = str(v).trim();
   454→  return s.length ? s : undefined;
   455→}
   456→
   457→function num(v: unknown, fallback: number): number {
   458→  const n = Number(v);
   459→  return Number.isFinite(n) ? n : fallback;
   460→}
   461→
   462→function bool(v: unknown): boolean | undefined {
   463→  return typeof v === "boolean" ? v : undefined;
   464→}

   677→export type HeroSlideRow = {
   678→  id: string; scope: string; sort: number; kind: "image" | "youtube";
   679→  src: string; alt?: string; active: boolean;
   680→};
   681→
   682→export function mapHeroSlide(row: Record<string, unknown>): HeroSlideRow {
   683→  const src = str(row.src);
   684→  let kind: "image" | "youtube" = str(row.kind, "image") === "youtube" ? "youtube" : "image";
   685→  if (kind === "image" && parseYouTubeId(src)) kind = "youtube";
   686→  return {
   687→    id: str(row.id),
   688→    scope: str(row.scope, "home"),
   689→    sort: num(row.sort, 0),
   690→    kind,
   691→    src,
   692→    alt: ostr(row.alt),
   693→    active: row.active === undefined ? true : Boolean(row.active),
   694→  };
   695→}
   696→
   697→export function heroSlideRowToSlide(row: HeroSlideRow): HeroSlide {
   698→  if (row.kind === "youtube") {
   699→    const id = parseYouTubeId(row.src);
   700→    if (id) return { type: "youtube", id };
   701→    return { type: "image", src: row.src, alt: row.alt ?? "" };
   702→  }
   703→  return { type: "image", src: row.src, alt: row.alt ?? "" };
   704→}
   705→
   706→export function buildHomeHeroSlides(rows: HeroSlideRow[], videoUrls: string[]): HeroSlide[] {
   707→  const images = rows
   708→    .filter((r) => r.active && r.kind === "image")
   709→    .sort((a, b) => a.sort - b.sort)
   710→    .map(heroSlideRowToSlide);
   711→  const dbVideos = rows
   712→    .filter((r) => r.active && r.kind === "youtube")
   713→    .sort((a, b) => a.sort - b.sort)
   714→    .map(heroSlideRowToSlide)
   715→    .filter((s) => s.type === "youtube");
   716→  const settingsVideos = videoUrls
   717→    .map(parseYouTubeId)
   718→    .filter((id): id is string => Boolean(id))
   719→    .map((id) => ({ type: "youtube", id }) as HeroSlide);
   720→  const slides = buildHeroSlides(images, [...dbVideos, ...settingsVideos]);
   721→  return slides.length ? slides : buildHeroSlides(homeHeroImages, settingsVideos);
   722→}
