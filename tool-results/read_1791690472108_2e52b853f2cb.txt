    46→function localizeReadTime(readTime: string, bn: boolean, minRead: string) {
    47→  if (!bn) return readTime;
    48→  const n = readTime.match(/\d+/)?.[0] ?? "";
    49→  return n ? `${toBnDigits(n)} ${minRead}` : readTime;

   395→function ReadingProgress() {
   396→  const [progress, setProgress] = useState(0);
   397→
   398→  useEffect(() => {
   399→    let raf = 0;
   400→    const update = () => {
   401→      raf = 0;
   402→      const el = document.documentElement;
   403→      const total = el.scrollHeight - el.clientHeight;
   404→      setProgress(total > 0 ? Math.min(1, el.scrollTop / total) : 0);
   405→    };
   406→    const onScroll = () => {
   407→      if (!raf) raf = requestAnimationFrame(update);
   408→    };
   409→    update();
   410→    window.addEventListener("scroll", onScroll, { passive: true });
   411→    window.addEventListener("resize", onScroll, { passive: true });
   412→    return () => {
   413→      window.removeEventListener("scroll", onScroll);
   414→      window.removeEventListener("resize", onScroll);
   415→      if (raf) cancelAnimationFrame(raf);
   416→    };
   417→  }, []);
   418→
   419→  return (
   420→    <div
   421→      className="reading-progress"
   422→      style={{ transform: `scaleX(${progress})` }}
   423→      aria-hidden="true"
   424→    />
   425→  );
   426→}

   428→/* Active-section tracking for the article table of contents */
   429→function useActiveSection(count: number) {
   430→  const [active, setActive] = useState(0);
   431→
   432→  useEffect(() => {
   433→    if (count < 2) return;
   434→    const headings = Array.from({ length: count }, (_, i) =>
   435→      document.getElementById(`section-${i}`)
   436→    );
   437→    const visible = headings.filter(Boolean) as HTMLElement[];
   438→    if (visible.length === 0) return;
   439→    const observer = new IntersectionObserver(
   440→      (entries) => {
   441→        entries.forEach((entry) => {
   442→          if (entry.isIntersecting) {
   443→            const idx = Number(entry.target.id.replace("section-", ""));
   444→            setActive(idx);
   445→          }
   446→        });
   447→      },
   448→      { rootMargin: "-25% 0px -65% 0px", threshold: 0 }
   449→    );
   450→    visible.forEach((h) => observer.observe(h));
   451→    return () => observer.disconnect();
   452→  }, [count]);
   453→  return [active, setActive] as const;
   454→}

   456→export function BlogDetailPage({ slug }: { slug: string }) {
   457→  const { navigate } = useRouter();
   458→  const { lang, d } = useLang();
   459→  const bn = lang === "bn";
   460→  const t = d.blogPage;
   461→  const post = getPost(slug);
   462→  const [activeSection, setActiveSection] = useActiveSection(post?.sections.length ?? 0);
   463→
   464→  const goToSection = (i: number) => {
   465→    /* NOTE: never use href="#section-x" — it would hijack the hash router */
   466→    setActiveSection(i);
   467→    const el = document.getElementById(`section-${i}`);
   468→    el?.scrollIntoView({ behavior: "smooth", block: "start" });
   469→    el?.focus({ preventScroll: true });
   470→  };
   471→
   472→  const { toast } = useToast();
   473→  const [copied, setCopied] = useState(false);
   474→
   475→  /* Deep-linkable article URL (matches the canonical production URL) */
   476→  const articleUrl = () => `${window.location.origin}/#/blog/${slug}`;
   477→  const shareTitle = () => (bn && post?.titleBn ? post.titleBn : post?.title ?? "Dental health article");
   478→
   479→  const shareWhatsApp = () => {
   480→    const text = `${shareTitle()} — ${articleUrl()}`;

   487→      "_blank",

   493→    try {
   494→      await navigator.clipboard.writeText(articleUrl());
   495→      setCopied(true);
   496→      toast({
   497→        title: t.shareCopied,
