    45→/* "8 min read" → "৮ মিনিট পাঠ" when Bangla (mirrors blog page) */
    46→function localizeReadTime(readTime: string, bn: boolean, minRead: string) {
    47→  if (!bn) return readTime;
    48→  const n = readTime.match(/\d+/)?.[0] ?? "";
    49→  return n ? `${toBnDigits(n)} ${minRead}` : readTime;
    50→}
