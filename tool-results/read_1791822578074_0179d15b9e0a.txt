    85→function clone<T>(value: T): T {
    86→  return JSON.parse(JSON.stringify(value)) as T;
    87→}
    88→
    89→/** Drop empty optional Bangla twins (…Bn keys) so the site falls back to EN. */
    90→function pruneBangla<T>(value: T): T {
    91→  if (Array.isArray(value)) return value.map((item) => pruneBangla(item)) as unknown as T;
    92→  if (value !== null && typeof value === "object") {
    93→    const out: Record<string, unknown> = {};
    94→    for (const [key, v] of Object.entries(value as Record<string, unknown>)) {
    95→      if (typeof v === "string" && v.trim() === "" && key.endsWith("Bn")) continue;
    96→      out[key] = pruneBangla(v);
    97→    }
    98→    return out as T;
    99→  }
   100→  return value;
   101→}

   244→export function AboutPanel({ onSessionExpired }: PanelProps) {
   245→  const { get, post, error, clearError } = useAdminFetch(onSessionExpired);
   246→  const { ok, fail } = useAdminToast();
   247→  const { refresh } = useContent();
   248→
   249→  const [snapshot, setSnapshot] = useState<AboutContent | null>(null);
   250→  const [form, setForm] = useState<AboutContent | null>(null);
   251→  const [loadFailed, setLoadFailed] = useState(false);
   252→  const [loadTick, setLoadTick] = useState(0);
   253→  const [saving, setSaving] = useState(false);
   254→
   255→  /* Initial fetch (+ manual retry). The async IIFE keeps every setState
   256→     behind an await so the effect stays purely a subscription. */
   257→  useEffect(() => {
   258→    let cancelled = false;
   259→    (async () => {
   260→      const data = await get<{ about: AboutContent }>("/api/admin/pages");
   261→      if (cancelled) return;
   262→      if (!data) {
   263→        setLoadFailed(true);
   264→        return;
   265→      }
   266→      setLoadFailed(false);
   267→      setSnapshot(clone(data.about));
   268→      setForm(clone(data.about));
   269→    })();
   270→    return () => {
   271→      cancelled = true;
   272→    };
   273→  }, [get, loadTick]);
   274→
   275→  const dirty = useMemo(
   276→    () => (snapshot && form ? JSON.stringify(form) !== JSON.stringify(snapshot) : false),
   277→    [snapshot, form]
   278→  );
   279→
   280→  const groupDirty = useCallback(
   281→    (pick: (a: AboutContent) => unknown) =>
   282→      snapshot && form ? JSON.stringify(pick(form)) !== JSON.stringify(pick(snapshot)) : false,
   283→    [snapshot, form]
   284→  );
   285→
   286→  async function save() {
   287→    if (!form || saving) return;
   288→    setSaving(true);
   289→    const payload = pruneBangla(form);
   290→    const res = await post<{ error?: string }>("/api/admin/pages", {
   291→      key: "about",
   292→      data: payload,
   293→    });
   294→    setSaving(false);
   295→    if (res.ok) {
   296→      setSnapshot(payload);
   297→      setForm(clone(payload));
   298→      ok("About page saved", "The live site now shows your changes.");
   299→      await refresh();
   300→    } else {
   301→      fail("Could not save the about page", res.data?.error ?? "Please try again.");
   302→    }
   303→  }
   304→
   305→  function reset() {
   306→    if (snapshot) setForm(clone(snapshot));
   307→    clearError();
   308→  }
   309→
   310→  /* ---- generic field setter ---- */
   311→  const set = <K extends keyof AboutContent>(key: K, value: AboutContent[K]) =>
   312→    setForm((f) => (f ? { ...f, [key]: value } : f));
   313→
   314→  /* ---- timeline ---- */
   315→  const setTimelineItem = (
   316→    index: number,
   317→    patch: Partial<AboutContent["timeline"][number]>
   318→  ) =>
   319→    setForm((f) =>
   320→      f
   321→        ? {
   322→            ...f,
   323→            timeline: f.timeline.map((t, i) => (i === index ? { ...t, ...patch } : t)),
   324→          }
   325→        : f
   326→    );
   327→
   328→  const addTimelineItem = () =>
   329→    setForm((f) =>
   330→      f && f.timeline.length < MAX_TIMELINE
   331→        ? { ...f, timeline: [...f.timeline, { title: "", description: "" }] }
   332→        : f
   333→    );
   334→
   335→  const removeTimelineItem = (index: number) =>
   336→    setForm((f) => (f ? { ...f, timeline: f.timeline.filter((_, i) => i !== index) } : f));
   337→
   338→  /* ---- founder items ---- */
   339→  const setFounderItem = (
   340→    index: number,
   341→    patch: Partial<AboutContent["founderItems"][number]>
   342→  ) =>
   343→    setForm((f) =>
   344→      f
   345→        ? {
   346→            ...f,
   347→            founderItems: f.founderItems.map((item, i) =>
   348→              i === index ? { ...item, ...patch } : item
   349→            ),
   350→          }
   351→        : f
   352→    );
   353→
   354→  const addFounderItem = () =>
   355→    setForm((f) =>
   356→      f && f.founderItems.length < MAX_FOUNDER_ITEMS
   357→        ? { ...f, founderItems: [...f.founderItems, { title: "", text: "" }] }
   358→        : f
   359→    );
   360→
   361→  const removeFounderItem = (index: number) =>
   362→    setForm((f) => (f ? { ...f, founderItems: f.founderItems.filter((_, i) => i !== index) } : f));
   363→
   364→  /* ---- values ---- */
   365→  const setValue = (index: number, patch: Partial<AboutContent["values"][number]>) =>
   366→    setForm((f) =>
   367→      f
   368→        ? { ...f, values: f.values.map((v, i) => (i === index ? { ...v, ...patch } : v)) }
   369→        : f
   370→    );
   371→
   372→  const addValue = () =>
   373→    setForm((f) =>
   374→      f && f.values.length < MAX_VALUES
   375→        ? { ...f, values: [...f.values, { icon: "BadgeCheck", title: "", description: "" }] }
   376→        : f
   377→    );
   378→
   379→  const removeValue = (index: number) =>
   380→    setForm((f) => (f ? { ...f, values: f.values.filter((_, i) => i !== index) } : f));
   381→
   382→  /* ---- render ---- */
   383→
   384→  if (!snapshot || !form) {
   385→    if (loadFailed) {
   386→      return (
   387→        <div className="space-y-4">
   388→          <PanelHeader title="About Page Content" />
   389→          <ErrorNote message={error ?? "Could not load the about page content."} />
   390→          <div>
   391→            <Button
   392→              onClick={() => {
   393→                setLoadFailed(false);
   394→                clearError();
   395→                setLoadTick((t) => t + 1);
   396→              }}
   397→              variant="outline"
   398→              size="sm"
   399→              className="rounded-full border-blue-200 text-brand-navy hover:bg-blue-50"
   400→            >
   401→              <RefreshCw className="h-4 w-4" aria-hidden="true" />
   402→              Try again
   403→            </Button>
   404→          </div>
   405→        </div>
   406→      );
   407→    }
   408→    return <PanelSkeleton />;
   409→  }
   410→
   411→  return (
   412→    <div className="space-y-4">
   413→      <PanelHeader
   414→        title="About Page Content"
   415→        description="Story, mission & vision, journey timeline, founder, values and the closing CTA — saves go live instantly."
   416→      />
   417→      {error ? <ErrorNote message={error} /> : null}
   418→
   419→      {/* ============ PAGE HEAD ============ */}
   420→      <FormSection
   421→        icon={Landmark}
   422→        title="Page Head"
   423→        description="Eyebrow, main title and intro paragraph"
   424→        edited={groupDirty((a) => [a.eyebrow, a.title, a.description])}
   425→        defaultOpen
   426→      >
   427→        <BilingualField
   428→          label="Eyebrow"
   429→          enValue={form.eyebrow}
   430→          bnValue={form.eyebrowBn}
   431→          onEnChange={(v) => set("eyebrow", v)}
   432→          onBnChange={(v) => set("eyebrowBn", v)}
   433→        />
   434→        <BilingualField
   435→          label="Title"
   436→          enValue={form.title}
   437→          bnValue={form.titleBn}
   438→          onEnChange={(v) => set("title", v)}
   439→          onBnChange={(v) => set("titleBn", v)}
   440→        />
   441→        <BilingualField
   442→          label="Description"
   443→          textarea
   444→          rows={4}
   445→          enValue={form.description}
   446→          bnValue={form.descriptionBn}
   447→          onEnChange={(v) => set("description", v)}
   448→          onBnChange={(v) => set("descriptionBn", v)}
   449→        />
   450→      </FormSection>
   451→
   452→      {/* ============ STORY ============ */}
   453→      <FormSection
   454→        icon={BookOpenText}
   455→        title="Story"
   456→        description="Founding-year badge, pull-quote and second paragraph"
   457→        edited={groupDirty((a) => [a.storyBadge, a.storyQuote, a.storyP2])}
   458→      >
   459→        <BilingualField
   460→          label="Story badge"
   461→          enValue={form.storyBadge}
   462→          bnValue={form.storyBadgeBn}
   463→          onEnChange={(v) => set("storyBadge", v)}
   464→          onBnChange={(v) => set("storyBadgeBn", v)}
   465→        />
   466→        <BilingualField
   467→          label="Pull quote"
   468→          textarea
   469→          rows={3}
   470→          enValue={form.storyQuote}

   472→          onEnChange={(v) => set("storyQuote", v)}
