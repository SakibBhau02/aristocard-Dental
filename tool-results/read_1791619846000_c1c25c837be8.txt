   379→   379→
   380→   380→function ReadingProgress() {
   381→   381→  const [progress, setProgress] = useState(0);
   382→   382→
   383→   383→  useEffect(() => {
   384→   384→    let raf = 0;
   385→   385→    const update = () => {
   386→   386→      raf = 0;
   387→   387→      const el = document.documentElement;
   388→   388→      const total = el.scrollHeight - el.clientHeight;
   389→   389→      setProgress(total > 0 ? Math.min(1, el.scrollTop / total) : 0);
   390→   390→    };
   391→   391→    const onScroll = () => {
   392→   392→      if (!raf) raf = requestAnimationFrame(update);
   393→   393→    };
   394→   394→    update();
   395→   395→    window.addEventListener("scroll", onScroll, { passive: true });
   396→   396→    window.addEventListener("resize", onScroll, { passive: true });
   397→   397→    return () => {
   398→   398→      window.removeEventListener("scroll", onScroll);
   399→   399→      window.removeEventListener("resize", onScroll);
   400→   400→      if (raf) cancelAnimationFrame(raf);
   401→   401→    };
   402→   402→  }, []);
   403→   403→
   404→   404→  return (
   405→   405→    <div
   406→   406→      className="reading-progress"
   407→   407→      style={{ transform: `scaleX(${progress})` }}
   408→   408→      aria-hidden="true"
   409→   409→    />
   410→   410→  );
   411→   411→}
