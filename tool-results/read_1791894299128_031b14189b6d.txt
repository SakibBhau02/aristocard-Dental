   277→/** Live clock re-renders once a minute for the session countdown. */
   278→function useMinuteNow(): number {
   279→  const [now, setNow] = useState(() => Date.now());
   280→  useEffect(() => {
   281→    const id = window.setInterval(() => setNow(Date.now()), 60_000);
   282→    return () => window.clearInterval(id);
   283→  }, []);
   284→  return now;
   285→}
   286→
   287→/** "7h 42m" / "42m" / "less than a minute". */
   288→function formatSessionRemaining(ms: number): string {
   289→  const totalMin = Math.max(0, Math.round(ms / 60_000));
   290→  const h = Math.floor(totalMin / 60);
   291→  const m = totalMin % 60;
   292→  if (h > 0) return `${h}h ${m}m`;
   293→  if (totalMin > 0) return `${m}m`;
   294→  return "less than a minute";
   295→}

   301→export function AdminPage() {
   302→  usePageMeta({
   303→    title: "Admin — Staff Dashboard",
   304→    description: "Staff-only clinic management dashboard.",
   305→  });
   306→
   307→  /* noindex for this back-office page */
   308→  useEffect(() => {
   309→    const meta = document.createElement("meta");
   310→    meta.name = "robots";
   311→    meta.content = "noindex, nofollow";
   312→    meta.dataset.adminNoindex = "true";
   313→    document.head.appendChild(meta);
   314→    return () => {
   315→      meta.remove();
   316→    };
   317→  }, []);
   318→
   319→  const [authState, setAuthState] = useState<"checking" | "locked" | "open">(
   320→    "checking"
   321→  );
   322→  const [sessionUser, setSessionUser] = useState<AdminSessionUser | null>(null);
   323→  const [sessionExpiresAt, setSessionExpiresAt] = useState<number | null>(null);
   324→  const [gateNotice, setGateNotice] = useState<string | null>(null);
   325→
   326→  /* Restore an existing cookie session (and clear any legacy
   327→     pre-session-era key). The identity travels in the httpOnly cookie;
   328→     the password is never stored client-side. */
   329→  useEffect(() => {
   330→    try {
   331→      sessionStorage.removeItem(LEGACY_SESSION_KEY);
   332→    } catch {
   333→      /* ignore */
   334→    }
   335→    let cancelled = false;
   336→    (async () => {
   337→      try {
   338→        const res = await fetch("/api/admin/session");
   339→        const data = (await res.json().catch(() => null)) as
   340→          | { authenticated?: boolean; user?: AdminSessionUser; expiresAt?: number | null }
   341→          | null;
   342→        if (!cancelled && res.ok && data?.authenticated && data.user) {
   343→          setSessionUser(data.user);
   344→          setSessionExpiresAt(typeof data.expiresAt === "number" ? data.expiresAt : null);
   345→          setAuthState("open");
   346→          return;
   347→        }
   348→        if (!cancelled) setAuthState("locked");
   349→      } catch {
   350→        if (!cancelled) setAuthState("locked");
   351→      }
   352→    })();
   353→    return () => {
   354→      cancelled = true;
   355→    };
   356→  }, []);
   357→
   358→  const handleUnlock = useCallback((session: SessionInfo) => {
   359→    setSessionUser(session.user);
   360→    setSessionExpiresAt(session.expiresAt);
   361→    setGateNotice(null);
   362→    setAuthState("open");
   363→  }, []);
   364→
   365→  const handleLogout = useCallback(async () => {
   366→    try {
   367→      await fetch("/api/admin/logout", { method: "POST" });
   368→    } catch {
   369→      /* clearing the local state matters more than the network call */
   370→    }
   371→    setSessionUser(null);
   372→    setSessionExpiresAt(null);
   373→    setGateNotice(null);
   374→    setAuthState("locked");
   375→  }, []);
   376→
   377→  /* Mid-session expiry: back to the gate with a notice (no dead panel). */
   378→  const handleSessionExpired = useCallback(() => {
   379→    setSessionUser(null);
   380→    setSessionExpiresAt(null);
   381→    setGateNotice("Session expired — sign in again to continue.");
   382→    setAuthState("locked");
   383→  }, []);
   384→
   385→  if (authState === "checking") return <AdminBoot />;
   386→
   387→  return authState === "open" && sessionUser ? (
   388→    <AdminShell
   389→      user={sessionUser}
   390→      sessionExpiresAt={sessionExpiresAt}
   391→      onLogout={handleLogout}
   392→      onSessionExpired={handleSessionExpired}
   393→    />
   394→  ) : (
   395→    <AdminGate onUnlock={handleUnlock} notice={gateNotice} />
   396→  );
   397→}

   432→function AdminGate({
   433→  onUnlock,
   434→  notice,
   435→}: {
   436→  onUnlock: (session: SessionInfo) => void;
   437→  notice?: string | null;
   438→}) {
   439→  const [username, setUsername] = useState("");
   440→  const [password, setPassword] = useState("");
   441→  const [error, setError] = useState<string | null>(null);
   442→  const [checking, setChecking] = useState(false);
   443→  const usernameRef = useRef<HTMLInputElement | null>(null);
   444→
   445→  /* focus management: land on the username field on every gate mount */
   446→  useEffect(() => {
   447→    usernameRef.current?.focus();
   448→  }, []);
   449→
   450→  async function handleSubmit(e: React.FormEvent) {
   451→    e.preventDefault();
   452→    if (checking) return;
   453→    setError(null);
   454→    if (!username.trim() || !password) {
   455→      setError("Enter both your username and password.");
   456→      return;
   457→    }
   458→    setChecking(true);
   459→    try {
   460→      /* The password travels once, in this request body — the server
   461→         responds with an httpOnly session cookie that does the rest. */
   462→      const res = await fetch("/api/admin/login", {
   463→        method: "POST",
   464→        headers: { "Content-Type": "application/json" },
   465→        body: JSON.stringify({ username: username.trim(), password }),
   466→      });
   467→      const data = (await res.json().catch(() => null)) as
   468→        | { user?: AdminSessionUser; expiresAt?: number | null; error?: string }
   469→        | null;
   470→      if (res.status === 401) {
   471→        setError(data?.error ?? "Incorrect username or password.");
   472→        return;
   473→      }
   474→      if (res.status === 429) {
   475→        setError(data?.error ?? "Too many attempts. Please wait a few minutes and try again.");
   476→        return;
   477→      }
   478→      if (!res.ok || !data?.user) {
   479→        setError(data?.error ?? "The sign-in request failed. Please try again.");
   480→        return;
   481→      }
   482→      onUnlock({
   483→        user: data.user,
   484→        expiresAt: typeof data.expiresAt === "number" ? data.expiresAt : null,
   485→      });
   486→    } catch {
   487→      setError("Could not reach the server. Check your connection and try again.");
   488→    } finally {
   489→      setChecking(false);
   490→    }
   491→  }

   514→            Aristocrat Dental Care

  1070→/** "Session expires in 7h 42m" — amber + re-sign-in hint under 10 min. */
  1071→function SessionCountdown({ remainingMs }: { remainingMs: number | null }) {
  1072→  if (remainingMs === null) return null;
  1073→
  1074→  if (remainingMs <= 0) {
  1075→    return (
  1076→      <p className="mt-2.5 flex items-center gap-1.5 text-[11px] font-semibold text-amber-300">
  1077→        <Timer className="h-3.5 w-3.5 shrink-0" aria-hidden="true" />
  1078→        Session expired — sign in again
  1079→      </p>
  1080→    );
  1081→  }
  1082→
  1083→  const text = formatSessionRemaining(remainingMs);
  1084→  const warning = remainingMs < 10 * 60_000;
  1085→  return (
  1086→    <p
  1087→      className={cn(
  1088→        "mt-2.5 flex items-center gap-1.5 text-[11px] font-medium",
  1089→        warning ? "text-amber-300" : "text-blue-200/50"
  1090→      )}
  1091→    >
  1092→      {warning ? (
  1093→        <Timer className="h-3.5 w-3.5 shrink-0" aria-hidden="true" />
  1094→      ) : (
  1095→        <Clock className="h-3.5 w-3.5 shrink-0" aria-hidden="true" />
  1096→      )}
  1097→      <span className="truncate">
  1098→        {warning
  1099→          ? `Session expires in ${text} — save your work and sign in again`
  1100→          : `Session expires in ${text}`}
  1101→      </span>
  1102→    </p>
  1103→  );
  1104→}
