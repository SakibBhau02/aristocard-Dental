   323→/* ------------------------------------------------------------------ */
   324→/*  Deep merge — DB sparse overrides over defaults                     */
   325→/* ------------------------------------------------------------------ */
   326→
   327→export function deepMerge<T>(base: T, override: unknown): T {
   328→  if (override === null || override === undefined) return base;
   329→  if (Array.isArray(base) || Array.isArray(override)) {
   330→    /* Arrays are replaced wholesale (never merged item-by-item). */
   331→    return (Array.isArray(override) ? override : base) as T;
   332→  }
   333→  if (typeof base === "object" && typeof override === "object" && base !== null) {
   334→    const out: Record<string, unknown> = { ...(base as Record<string, unknown>) };
   335→    for (const [k, v] of Object.entries(override as Record<string, unknown>)) {
   336→      const b = (base as Record<string, unknown>)[k];
   337→      out[k] = b !== undefined && b !== null && typeof b === "object" && !Array.isArray(b) && v !== null && typeof v === "object" && !Array.isArray(v)
   338→        ? deepMerge(b, v)
   339→        : v;
   340→    }
   341→    return out as T;
   342→  }
   343→  return (override as T) ?? base;
   344→}

   346→export function featuredServices(list: Service[], count = 4): Service[] {
   347→  const f = list.filter((s) => s.featured);

   355→function parseJson<T>(raw: string | null | undefined, fallback: T): T {
   356→  if (!raw) return fallback;
   357→  try {
   358→    const v = JSON.parse(raw);
   359→    return (v === null || v === undefined) ? fallback : (v as T);
   360→  } catch {
   361→    return fallback;
   362→  }
   363→}
   364→
   365→function str(v: unknown, fallback = ""): string {
   366→  return typeof v === "string" ? v : fallback;
   367→}
   368→
   369→function ostr(v: unknown): string | undefined {
   370→  const s = str(v).trim();
   371→  return s.length ? s : undefined;
   372→}
   373→
   374→function num(v: unknown, fallback: number): number {
   375→  const n = Number(v);
   376→  return Number.isFinite(n) ? n : fallback;
   377→}
   378→
   379→function bool(v: unknown): boolean | undefined {
   380→  return typeof v === "boolean" ? v : undefined;
   381→}

   413→    specializations: parseJson<string[]>(row.specializationsJson, []),

   415→    consultationDays: parseJson<string[]>(row.consultationDaysJson, []),

   422→/** DB DoctorRow → public Doctor type (as consumed by the site). */
   423→export function doctorRowToDoctor(row: DoctorRow): Doctor {
   424→  const numericId = row.legacyId || Math.abs(
   425→    Array.from(row.id).reduce((acc, ch) => (acc * 31 + ch.charCodeAt(0)) | 0, 7)
   426→  ) % 100000;
   427→  return {
   428→    slug: row.slug,
   429→    id: numericId,
   430→    name: row.name,
   431→    credentials: row.credentials,
   432→    role: row.role,
   433→    roleBn: row.roleBn,
   434→    department: row.department,
   435→    departmentBn: row.departmentBn,
   436→    bio: row.bio,
   437→    experienceYears: row.experienceYears,
   438→    patientsTreated: row.patientsTreated,
   439→    rating: row.rating,
   440→    specializations: row.specializations,
   441→    languages: row.languages,
   442→    consultationDays: row.consultationDays,
   443→    verified: true,
   444→    accent: row.accent,
   445→  };
   446→}

   594→export type HeroSlideRow = {
   595→  id: string; scope: string; sort: number; kind: "image" | "youtube";
   596→  src: string; alt?: string; active: boolean;
   597→};
   598→
   599→export function mapHeroSlide(row: Record<string, unknown>): HeroSlideRow {
   600→  const src = str(row.src);
   601→  let kind: "image" | "youtube" = str(row.kind, "image") === "youtube" ? "youtube" : "image";
   602→  if (kind === "image" && parseYouTubeId(src)) kind = "youtube";
   603→  return {
   604→    id: str(row.id),
   605→    scope: str(row.scope, "home"),
   606→    sort: num(row.sort, 0),
   607→    kind,
   608→    src,
   609→    alt: ostr(row.alt),
   610→    active: row.active === undefined ? true : Boolean(row.active),
   611→  };
   612→}
   613→
   614→export function heroSlideRowToSlide(row: HeroSlideRow): HeroSlide {
   615→  if (row.kind === "youtube") {
   616→    const id = parseYouTubeId(row.src);
   617→    if (id) return { type: "youtube", id };
   618→    return { type: "image", src: row.src, alt: row.alt ?? "" };
   619→  }
   620→  return { type: "image", src: row.src, alt: row.alt ?? "" };
   621→}
   622→
   623→export function buildHomeHeroSlides(rows: HeroSlideRow[], videoUrls: string[]): HeroSlide[] {
   624→  const images = rows
   625→    .filter((r) => r.active && r.kind === "image")
   626→    .sort((a, b) => a.sort - b.sort)
   627→    .map(heroSlideRowToSlide);
   628→  const dbVideos = rows
   629→    .filter((r) => r.active && r.kind === "youtube")
   630→    .sort((a, b) => a.sort - b.sort)
   631→    .map(heroSlideRowToSlide)
   632→    .filter((s) => s.type === "youtube");
   633→  const settingsVideos = videoUrls
   634→    .map(parseYouTubeId)
   635→    .filter((id): id is string => Boolean(id))
   636→    .map((id) => ({ type: "youtube", id }) as HeroSlide);
   637→  const slides = buildHeroSlides(images, [...dbVideos, ...settingsVideos]);
   638→  return slides.length ? slides : buildHeroSlides(homeHeroImages, settingsVideos);
   639→}
