   204→function toIsoDate(d: Date): string {
   205→  const p = (n: number) => String(n).padStart(2, "0");
   206→  return `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())}`;
   207→}
   208→
   209→export function localToday(): string {
   210→  return toIsoDate(new Date());

   213→function addDaysIso(iso: string, days: number): string {
   214→  const [y, m, d] = iso.split("-").map(Number);
   215→  return toIsoDate(new Date(y, m - 1, d + days));
   216→}

   237→/** "Oct 12, 2026" from an ISO date or full timestamp. */
   238→export function prettyDay(iso: string): string {
   239→  const d = new Date(iso);
   240→  if (Number.isNaN(d.getTime())) return iso;
   241→  return d.toLocaleDateString("en-US", {
   242→    month: "short",
   243→    day: "numeric",
   244→    year: "numeric",
   245→  });
   246→}
   247→
   248→/** "Oct 12, 2026 · 4:26 PM" from an ISO timestamp. */
   249→export function prettyStamp(iso: string): string {
   250→  const d = new Date(iso);
   251→  if (Number.isNaN(d.getTime())) return iso;
   252→  return `${prettyDay(iso)} · ${d.toLocaleTimeString("en-US", {
   253→    hour: "numeric",
   254→    minute: "2-digit",
   255→  })}`;
   256→}
   257→
   258→/** "3 days ago" style relative label (best-effort, local clock). */
   259→export function relativeTime(iso: string): string {
   260→  const then = new Date(iso).getTime();
   261→  if (Number.isNaN(then)) return "";
   262→  const mins = Math.round((Date.now() - then) / 60_000);
   263→  if (mins < 1) return "just now";
   264→  if (mins < 60) return `${mins} min ago`;
   265→  const hours = Math.round(mins / 60);
   266→  if (hours < 24) return `${hours} hr ago`;
   267→  const days = Math.round(hours / 24);
   268→  if (days < 7) return `${days} day${days === 1 ? "" : "s"} ago`;
   269→  const weeks = Math.round(days / 7);
   270→  if (weeks < 5) return `${weeks} wk ago`;
   271→  const months = Math.round(days / 30);
   272→  if (months < 12) return `${months} mo ago`;
   273→  return `${Math.round(months / 12)} yr ago`;
   274→}

   276→/** "Mon 29" chart label from a YYYY-MM-DD date. */
   277→export function chartDayLabel(iso: string): string {
   278→  const [y, m, d] = iso.split("-").map(Number);
   279→  if (!y || !m || !d) return iso;
   280→  const dt = new Date(Date.UTC(y, m - 1, d));
   281→  const weekday = dt.toLocaleDateString("en-US", {
   282→    timeZone: "UTC",
   283→    weekday: "short",
   284→  });
   285→  return `${weekday} ${d}`;
   286→}

   288→/** Avatar initials, e.g. "Demo Patient" → "DP". */
   289→export function initialsOf(name: string): string {
   290→  const parts = name.trim().split(/\s+/).filter(Boolean);
   291→  if (parts.length === 0) return "?";
   292→  return parts
   293→    .slice(0, 2)
   294→    .map((p) => p.charAt(0).toUpperCase())
   295→    .join("");
   296→}

   298→/** SVG rect path with rounded TOP corners only (for the trend bars). */
   299→export function topRoundedBarPath(
   300→  x: number,
   301→  y: number,
   302→  w: number,
   303→  h: number,
   304→  r: number
   305→): string {
   306→  const rr = Math.min(r, h / 2, w / 2);
   307→  return [
   308→    `M ${x} ${y + h}`,
   309→    `L ${x} ${y + rr}`,
   310→    `Q ${x} ${y} ${x + rr} ${y}`,
   311→    `L ${x + w - rr} ${y}`,
   312→    `Q ${x + w} ${y} ${x + w} ${y + rr}`,
   313→    `L ${x + w} ${y + h}`,
   314→    "Z",
   315→  ].join(" ");
   316→}

   318→/** Shared CSV download — the filename comes from Content-Disposition. */
   319→export async function downloadCsvFile(
   320→  path: string,
   321→  fallbackName: string
   322→): Promise<"ok" | "unauthorized" | "error"> {
   323→  try {
   324→    const res = await fetch(path);
   325→    if (res.status === 401) return "unauthorized";
   326→    if (!res.ok) return "error";
   327→    const blob = await res.blob();
   328→    const disposition = res.headers.get("Content-Disposition") ?? "";
   329→    const filename =
   330→      /filename="([^"]+)"/.exec(disposition)?.[1] ?? fallbackName;
   331→    const url = URL.createObjectURL(blob);
   332→    const a = document.createElement("a");
   333→    a.href = url;
   334→    a.download = filename;
   335→    document.body.appendChild(a);
   336→    a.click();
   337→    a.remove();
   338→    URL.revokeObjectURL(url);
   339→    return "ok";
   340→  } catch {
   341→    return "error";
   342→  }
   343→}

   396→/* ------------------------------------------------------------------ */
   397→/*  useAdminFetch — cookie-auth wrappers for the admin CRUD APIs        */
   398→/*  (same-origin fetch; 401 routes to onUnauthorized so the shell can   */
   399→/*   return to the login gate)                                          */
   400→/* ------------------------------------------------------------------ */
   401→
   402→export function useAdminFetch(onUnauthorized: () => void) {
   403→  const [loading, setLoading] = useState(false);
   404→  const [error, setError] = useState<string | null>(null);
   405→  const onAuthRef = useRef(onUnauthorized);
   406→  onAuthRef.current = onUnauthorized;
   407→
   408→  const handle401 = useCallback((res: Response): boolean => {
   409→    if (res.status === 401) {
   410→      onAuthRef.current();
   411→      return true;
   412→    }
   413→    return false;
   414→  }, []);
   415→
   416→  const get = useCallback(
   417→    async <T,>(path: string): Promise<T | null> => {
   418→      setLoading(true);
   419→      setError(null);
   420→      try {
   421→        const res = await fetch(path);
   422→        if (handle401(res)) return null;
   423→        const data = await res.json().catch(() => null);
   424→        if (!res.ok) {
   425→          throw new Error(
   426→            (data as { error?: string } | null)?.error ?? `Request failed (${res.status}).`
   427→          );
   428→        }
   429→        return data as T;
   430→      } catch (err) {
   431→        setError(err instanceof Error ? err.message : "Request failed.");
   432→        return null;
   433→      } finally {
   434→        setLoading(false);
   435→      }
   436→    },
   437→    [handle401]
   438→  );
   439→
   440→  const post = useCallback(
   441→    async <T,>(
   442→      path: string,
   443→      payload: unknown
   444→    ): Promise<{ ok: boolean; data: T | null; status: number }> => {
   445→      setLoading(true);
   446→      setError(null);
   447→      try {
   448→        const res = await fetch(path, {
   449→          method: "POST",
   450→          headers: { "Content-Type": "application/json" },
   451→          body: JSON.stringify(payload ?? {}),
   452→        });
   453→        if (handle401(res)) return { ok: false, data: null, status: 401 };
   454→        const data = await res.json().catch(() => null);
   455→        if (!res.ok) {
   456→          const message =
   457→            (data as { error?: string } | null)?.error ?? `Request failed (${res.status}).`;
   458→          setError(message);
   459→          return { ok: false, data: data as T | null, status: res.status };
   460→        }
   461→        return { ok: true, data: data as T, status: res.status };
   462→      } catch (err) {
   463→        setError(err instanceof Error ? err.message : "Request failed.");
   464→        return { ok: false, data: null, status: 0 };
   465→      } finally {
   466→        setLoading(false);
   467→      }
   468→    },
   469→    [handle401]
   470→  );
   471→
   472→  const del = useCallback(
   473→    async <T,>(path: string): Promise<{ ok: boolean; data: T | null; status: number }> => {
   474→      setLoading(true);
   475→      setError(null);
   476→      try {
   477→        const res = await fetch(path, { method: "DELETE" });
   478→        if (handle401(res)) return { ok: false, data: null, status: 401 };
   479→        const data = await res.json().catch(() => null);
   480→        if (!res.ok) {
   481→          const message =
   482→            (data as { error?: string } | null)?.error ?? `Request failed (${res.status}).`;
   483→          setError(message);
   484→          return { ok: false, data: data as T | null, status: res.status };
   485→        }
   486→        return { ok: true, data: data as T, status: res.status };
   487→      } catch (err) {
   488→        setError(err instanceof Error ? err.message : "Request failed.");
   489→        return { ok: false, data: null, status: 0 };
   490→      } finally {
   491→        setLoading(false);
   492→      }
   493→    },
   494→    [handle401]
   495→  );
   496→
   497→  const clearError = useCallback(() => setError(null), []);
   498→
   499→  return { get, post, del, loading, error, handle401, clearError };
   500→}
