    41→function localizeReadTime(readTime: string, bn: boolean, minRead: string) {
    42→  if (!bn) return readTime;
    43→  const n = readTime.match(/\d+/)?.[0] ?? "";
    44→  return n ? `${toBnDigits(n)} ${minRead}` : readTime;
    45→}

   390→function ReadingProgress() {
   391→  const [progress, setProgress] = useState(0);
   392→
   393→  useEffect(() => {
   394→    let raf = 0;
   395→    const update = () => {
   396→      raf = 0;
   397→      const el = document.documentElement;
   398→      const total = el.scrollHeight - el.clientHeight;
   399→      setProgress(total > 0 ? Math.min(1, el.scrollTop / total) : 0);
   400→    };
   401→    const onScroll = () => {
   402→      if (!raf) raf = requestAnimationFrame(update);
   403→    };
   404→    update();
   405→    window.addEventListener("scroll", onScroll, { passive: true });
   406→    window.addEventListener("resize", onScroll, { passive: true });
   407→    return () => {
   408→      window.removeEventListener("scroll", onScroll);
   409→      window.removeEventListener("resize", onScroll);
   410→      if (raf) cancelAnimationFrame(raf);
   411→    };
   412→  }, []);
   413→
   414→  return (
   415→    <div
   416→      className="reading-progress"
   417→      style={{ transform: `scaleX(${progress})` }}
   418→      aria-hidden="true"
   419→    />
   420→  );
   421→}

   423→/* Active-section tracking for the article table of contents */
   424→function useActiveSection(count: number) {
   425→  const [active, setActive] = useState(0);
   426→
   427→  useEffect(() => {
   428→    if (count < 2) return;
   429→    const headings = Array.from({ length: count }, (_, i) =>
   430→      document.getElementById(`section-${i}`)
   431→    );
   432→    const visible = headings.filter(Boolean) as HTMLElement[];
   433→    if (visible.length === 0) return;
   434→    const observer = new IntersectionObserver(
   435→      (entries) => {
   436→        entries.forEach((entry) => {
   437→          if (entry.isIntersecting) {
   438→            const idx = Number(entry.target.id.replace("section-", ""));
   439→            setActive(idx);
   440→          }
   441→        });
   442→      },
   443→      { rootMargin: "-25% 0px -65% 0px", threshold: 0 }
   444→    );
   445→    visible.forEach((h) => observer.observe(h));
   446→    return () => observer.disconnect();
   447→  }, [count]);
   448→  return [active, setActive] as const;
   449→}

   451→export function BlogDetailPage({ slug }: { slug: string }) {
