   306→/*  Reading progress bar — top of viewport while reading an article    */

   308→
   309→function ReadingProgress() {
   310→  const [progress, setProgress] = useState(0);
   311→
   312→  useEffect(() => {
   313→    let raf = 0;
   314→    const update = () => {
   315→      raf = 0;
   316→      const el = document.documentElement;
   317→      const total = el.scrollHeight - el.clientHeight;
   318→      setProgress(total > 0 ? Math.min(1, el.scrollTop / total) : 0);
   319→    };
   320→    const onScroll = () => {
   321→      if (!raf) raf = requestAnimationFrame(update);
   322→    };
   323→    update();
   324→    window.addEventListener("scroll", onScroll, { passive: true });
   325→    window.addEventListener("resize", onScroll, { passive: true });
   326→    return () => {
   327→      window.removeEventListener("scroll", onScroll);
   328→      window.removeEventListener("resize", onScroll);
   329→      if (raf) cancelAnimationFrame(raf);
   330→    };
   331→  }, []);
   332→
   333→  return (
   334→    <div
   335→      className="reading-progress"
   336→      style={{ transform: `scaleX(${progress})` }}
   337→      aria-hidden="true"
   338→    />
   339→  );
   340→}
