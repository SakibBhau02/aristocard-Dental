    91→function toIsoDate(d: Date): string {
    92→  const p = (n: number) => String(n).padStart(2, "0");
    93→  return `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())}`;
    94→}
    95→function localToday(): string {
    96→  return toIsoDate(new Date());
    97→}
    98→function addDaysIso(iso: string, days: number): string {
    99→  const [y, m, d] = iso.split("-").map(Number);
   100→  return toIsoDate(new Date(y, m - 1, d + days));
   101→}

   302→      try {
   303→        const res = await fetch("/api/admin/reviews", {

   330→          <p className="mt-2 text-sm text-amber-800">

   433→function AppointmentsPanel({
   434→  code,
   435→  onLogout,
   436→  onSessionExpired,
   437→}: {
   438→  code: string;
   439→  onLogout: () => void;
   440→  onSessionExpired: () => void;
   441→}) {
   442→  const [rows, setRows] = useState<AppointmentRow[]>([]);
   443→  const [stats, setStats] = useState<Stats | null>(null);
   444→  const [loading, setLoading] = useState(true);
   445→  const [loadError, setLoadError] = useState<string | null>(null);
   446→  const [filter, setFilter] = useState<StatusFilter>("ALL");
   447→  const [search, setSearch] = useState("");
   448→  const [busyRef, setBusyRef] = useState<string | null>(null);
   449→  const [lastUpdated, setLastUpdated] = useState<Date | null>(null);
   450→  const [preset, setPreset] = useState<DatePreset>("all");
   451→  const [fromDate, setFromDate] = useState("");
   452→  const [toDate, setToDate] = useState("");
   453→  const [range, setRange] = useState<{
   454→    from: string | null;
   455→    to: string | null;
   456→    label: string;
   457→    total: number;
   458→  } | null>(null);
   459→  const [exporting, setExporting] = useState(false);
   460→  const loadIdRef = useRef(0);
   461→
   462→  const load = useCallback(async () => {
   463→    const requestId = ++loadIdRef.current;
   464→    setLoading(true);
   465→    setLoadError(null);
   466→    try {
   467→      const qs = new URLSearchParams();
   468→      if (fromDate) qs.set("from", fromDate);
   469→      if (toDate) qs.set("to", toDate);
   470→      const query = qs.toString();
   471→      const res = await fetch(
   472→        `/api/admin/appointments${query ? `?${query}` : ""}`,
   473→        { headers: { "x-admin-code": code } }
   474→      );
   475→      if (res.status === 401) {
   476→        onSessionExpired();
   477→        return;
   478→      }
   479→      const data = await res.json();
   480→      if (!res.ok) throw new Error(data.error || "Failed to load");
   481→      if (requestId !== loadIdRef.current) return; /* a newer request is in flight */
   482→      setRows(data.appointments ?? []);
   483→      setStats(data.stats ?? null);
   484→      setRange(data.range ?? null);
   485→      setLastUpdated(new Date());
   486→    } catch (err) {
   487→      if (requestId !== loadIdRef.current) return;
   488→      setLoadError(err instanceof Error ? err.message : "Failed to load appointments.");
   489→    } finally {
   490→      if (requestId === loadIdRef.current) setLoading(false);
   491→    }
   492→  }, [code, fromDate, toDate, onSessionExpired]);
   493→
   494→  useEffect(() => {
   495→    load();
   496→  }, [load]);
   497→
   498→  /* auto-refresh every 60s (pause when tab hidden) */
   499→  useEffect(() => {
   500→    const id = window.setInterval(() => {
   501→      if (document.visibilityState === "visible") load();
   502→    }, 60_000);
   503→    return () => window.clearInterval(id);
   504→  }, [load]);
   505→
   506→  async function updateStatus(refCode: string, status: string) {
   507→    setBusyRef(refCode);
   508→    try {
   509→      const res = await fetch("/api/admin/appointments", {
   510→        method: "POST",
   511→        headers: {
   512→          "Content-Type": "application/json",
   513→          "x-admin-code": code,
   514→        },
   515→        body: JSON.stringify({ refCode, status }),
   516→      });
   517→      const data = await res.json();
   518→      if (res.status === 401) {
   519→        onSessionExpired();
   520→        return;
   521→      }
   522→      if (!res.ok) throw new Error(data.error || "Update failed");
   523→      setRows((prev) =>
   524→        prev.map((r) => (r.refCode === refCode ? { ...r, status } : r))
   525→      );
   526→      setStats((prev) => {
   527→        if (!prev) return prev;
   528→        const statusOf = (r: AppointmentRow) =>
   529→          r.refCode === refCode ? status : r.status;
   530→        const today = localToday();
   531→        return {
   532→          ...prev,
   533→          pending: rows.filter((r) => statusOf(r) === "PENDING").length,
   534→          confirmed: rows.filter((r) => statusOf(r) === "CONFIRMED").length,
   535→          completed: rows.filter((r) => statusOf(r) === "COMPLETED").length,
   536→          cancelled: rows.filter((r) => statusOf(r) === "CANCELLED").length,
   537→          today: rows.filter((r) => r.date === today && statusOf(r) !== "CANCELLED").length,
   538→          upcoming: rows.filter(
   539→            (r) => r.date >= today && (statusOf(r) === "PENDING" || statusOf(r) === "CONFIRMED")
   540→          ).length,
   541→        };
   542→      });
   543→    } catch {
   544→      setLoadError("Could not update status. Please retry.");
   545→    } finally {
   546→      setBusyRef(null);
   547→    }
   548→  }

   550→  /* --- date-range filtering --- */
   551→  function applyPreset(id: Exclude<DatePreset, "custom">) {
   552→    const p = DATE_PRESETS.find((d) => d.id === id) ?? DATE_PRESETS[0];
   553→    const [f, t] = p.range();
   554→    setPreset(id);
   555→    setFromDate(f);
   556→    setToDate(t);
   557→  }

   559→  function onFromDateChange(value: string) {

   562→  }

   564→  function onToDateChange(value: string) {
   565→    setToDate(value);

   571→  /* --- CSV export of the currently filtered set --- */
   572→  async function exportCsv() {
   573→    if (exporting) return;
   574→    setExporting(true);
   575→    try {
   576→      const qs = new URLSearchParams({ format: "csv" });
   577→      if (fromDate) qs.set("from", fromDate);
   578→      if (toDate) qs.set("to", toDate);
   579→      if (filter !== "ALL") qs.set("status", filter);
   580→      const res = await fetch(`/api/admin/appointments?${qs.toString()}`, {
   581→        headers: { "x-admin-code": code },
   582→      });
