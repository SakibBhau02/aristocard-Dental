   324→   324→/*  Deep merge — DB sparse overrides over defaults                     */
   325→   325→/* ------------------------------------------------------------------ */
   326→   326→
   327→   327→export function deepMerge<T>(base: T, override: unknown): T {
   328→   328→  if (override === null || override === undefined) return base;
   329→   329→  if (Array.isArray(base) || Array.isArray(override)) {
   330→   330→    /* Arrays are replaced wholesale (never merged item-by-item). */
   331→   331→    return (Array.isArray(override) ? override : base) as T;
   332→   332→  }
   333→   333→  if (typeof base === "object" && typeof override === "object" && base !== null) {
   334→   334→    const out: Record<string, unknown> = { ...(base as Record<string, unknown>) };
   335→   335→    for (const [k, v] of Object.entries(override as Record<string, unknown>)) {
   336→   336→      const b = (base as Record<string, unknown>)[k];
   337→   337→      out[k] = b !== undefined && b !== null && typeof b === "object" && !Array.isArray(b) && v !== null && typeof v === "object" && !Array.isArray(v)
   338→   338→        ? deepMerge(b, v)
   339→   339→        : v;
   340→   340→    }
   341→   341→    return out as T;
   342→   342→  }
   343→   343→  return (override as T) ?? base;
   344→   344→}

   355→   355→function parseJson<T>(raw: string | null | undefined, fallback: T): T {
   356→   356→  if (!raw) return fallback;
   357→   357→  try {
   358→   358→    const v = JSON.parse(raw);
   359→   359→    return (v === null || v === undefined) ? fallback : (v as T);
   360→   360→  } catch {
   361→   361→    return fallback;
   362→   362→  }
   363→   363→}

   365→   365→function str(v: unknown, fallback = ""): string {
   366→   366→  return typeof v === "string" ? v : fallback;
   367→   367→}
   368→   368→
   369→   369→function ostr(v: unknown): string | undefined {
   370→   370→  const s = str(v).trim();
   371→   371→  return s.length ? s : undefined;
   372→   372→}
   373→   373→
   374→   374→function num(v: unknown, fallback: number): number {
   375→   375→  const n = Number(v);
   376→   376→  return Number.isFinite(n) ? n : fallback;
   377→   377→}
   378→   378→
   379→   379→function bool(v: unknown): boolean | undefined {
   380→   380→  return typeof v === "boolean" ? v : undefined;
   381→   381→}
