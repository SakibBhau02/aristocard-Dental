    52→/* "8 min read" → "৮ মিনিট পাঠ" when Bangla */
    53→function localizeReadTime(readTime: string, bn: boolean, minRead: string) {
    54→  if (!bn) return readTime;
    55→  const n = readTime.match(/\d+/)?.[0] ?? "";
    56→  return n ? `${toBnDigits(n)} ${minRead}` : readTime;
    57→}
    58→
    59→/* "Dr. Shazia Khanam" → "SK" */
    60→function initialsOf(name: string) {
    61→  return name
    62→    .replace("Dr. ", "")
    63→    .split(" ")
    64→    .map((n) => n[0])
    65→    .slice(0, 2)
    66→    .join("");
    67→}
    68→
    69→/* Localized post date (bn-BD renders Bangla digits + month names) */
    70→function formatPostDate(
    71→  date: string,
    72→  bn: boolean,
    73→  opts?: Intl.DateTimeFormatOptions
    74→) {
    75→  return new Date(date).toLocaleDateString(
    76→    bn ? "bn-BD" : "en-GB",
    77→    opts ?? { day: "numeric", month: "short", year: "numeric" }
    78→  );
    79→}
    80→
    81→function localizeCount(bn: boolean, v: number) {
    82→  return bn ? toBnDigits(String(v)) : String(v);
    83→}

   205→function SidebarNewsletter() {
   206→  const { lang, d } = useLang();
   207→  const bn = lang === "bn";
   208→  const t = d.blogPage;
   209→  const [email, setEmail] = useState("");
   210→  const [state, setState] = useState<"idle" | "sending" | "done" | "error">(
   211→    "idle"
   212→  );
   213→
   214→  const submit = async (e: React.FormEvent) => {
   215→    e.preventDefault();
   216→    if (state === "sending") return;
   217→    setState("sending");
   218→    try {
   219→      const res = await fetch("/api/newsletter", {
   220→        method: "POST",
   221→        headers: { "Content-Type": "application/json" },
   222→        body: JSON.stringify({ email }),
   223→      });
   224→      setState(res.ok ? "done" : "error");
   225→    } catch {
   226→      setState("error");
   227→    }
   228→  };
   229→
   230→  return (
   231→    <div className="rounded-3xl border border-blue-100/80 bg-ice p-5">
   232→      <span className="inline-flex h-9 w-9 items-center justify-center rounded-full bg-white text-brand-blue ring-1 ring-blue-100">
   233→        <Mail className="h-4 w-4" aria-hidden="true" />
   234→      </span>
   235→      <h2 className="font-heading mt-3 text-base font-bold text-brand-navy">
   236→        {t.newsletterTitle}
   237→      </h2>
   238→      <p className="mt-1.5 text-xs leading-relaxed text-slate-500">
   239→        {t.newsletterText}
   240→      </p>
   241→      {state === "done" ? (
   242→        <div
   243→          className="mt-4 rounded-2xl bg-emerald-50 p-4 ring-1 ring-emerald-200"
   244→          role="status"
   245→        >
   246→          <p className="flex items-center gap-2 text-sm font-bold text-emerald-700">
   247→            <CheckCircle2 className="h-4 w-4 shrink-0" aria-hidden="true" />
   248→            {t.newsletterDone}
   249→          </p>
   250→          <p className="mt-1 text-xs leading-relaxed text-emerald-600">
   251→            {t.newsletterDoneDesc}
   252→          </p>
   253→        </div>
   254→      ) : (
   255→        <form onSubmit={submit} className="mt-4 flex flex-col gap-2">
   256→          <Input
   257→            type="email"
   258→            required
   259→            value={email}
   260→            onChange={(e) => {
   261→              setEmail(e.target.value);
   262→              if (state === "error") setState("idle");
   263→            }}
   264→            placeholder={t.newsletterPlaceholder}
   265→            aria-label={t.newsletterPlaceholder}
   266→            className="h-11 rounded-full border-blue-200/80 bg-white text-sm shadow-sm placeholder:text-slate-400 focus-visible:border-brand-blue"
   267→          />
   268→          <button
   269→            type="submit"
   270→            disabled={state === "sending"}
   271→            className="inline-flex h-11 items-center justify-center rounded-full bg-brand-navy text-xs font-bold text-white shadow-lg shadow-navy-900/20 transition-all duration-300 hover:bg-brand-blue focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-brand-blue focus-visible:ring-offset-2 disabled:opacity-60"
   272→          >
   273→            {state === "sending" ? "…" : t.newsletterCta}
   274→          </button>
   275→          {state === "error" && (
   276→            <p className="text-xs text-red-500" role="alert">
   277→              {t.newsletterFail} — {t.newsletterFailDesc}
   278→            </p>
   279→          )}
   280→        </form>
   281→      )}
   282→    </div>
   283→  );
   284→}

   290→const PAGE_SIZE_INITIAL = 9;

   293→export function BlogPage() {
   294→  const [category, setCategory] = useState("All");
   295→  const { query } = useRouter();
   296→  const { lang, d } = useLang();
   297→  const bn = lang === "bn";
   298→  const t = d.blogPage;
   299→  const { posts } = useBundle();
   300→
   301→  usePageMeta({
   302→    title: "Dental Health Blog — Expert Advice from Uttara's Trusted Dentists",
   303→    description:
   304→      "Evidence-based dental health guides from Aristocrat Dental Care specialists: checkup frequency, root canal facts, Invisalign vs braces, kids' dental care & more.",
   305→    jsonLd: {
   306→      "@context": "https://schema.org",
   307→      "@type": "Blog",
   308→      name: "Aristocrat Dental Care Health Blog",
   309→      url: "https://aristocratdentalcarebd.com/blog",
   310→      publisher: { "@type": "Organization", name: "Aristocrat Dental Care & Orthodontic Center" },
   311→    },
   312→  });
   313→
   314→  /* ?q= deep-link prefill (tag chips on article pages link here).
   315→     The typed value is keyed against the URL query so navigating to
   316→     /blog?q=… adopts it instantly, while typing never rewrites the URL. */
   317→  const qParam = query.get("q") ?? "";
   318→  const [searchState, setSearchState] = useState({ q: qParam, value: qParam });
   319→  const search = searchState.q === qParam ? searchState.value : qParam;
   320→  const setSearch = (v: string) => setSearchState({ q: qParam, value: v });
   321→
   322→  const searching = search.trim().length > 0;
   323→
   324→  /* Category chips — derived from the live posts (same as the static list) */
   325→  const blogCategories = useMemo(
   326→    () => ["All", ...Array.from(new Set(posts.map((p) => p.category)))],
   327→    [posts]
   328→  );
   329→
   330→  /* Category filter combined with text search (EN + BN content + tags) */
   331→  const filtered = useMemo(() => {
   332→    const q = search.trim().toLowerCase();
   333→    return posts.filter((p) => {
   334→      const matchCategory = category === "All" || p.category === category;
   335→      const matchSearch =
   336→        !q ||
   337→        p.title.toLowerCase().includes(q) ||
   338→        p.excerpt.toLowerCase().includes(q) ||
   339→        p.category.toLowerCase().includes(q) ||
   340→        p.tags.some((tag) => tag.toLowerCase().includes(q)) ||
   341→        (p.titleBn?.toLowerCase().includes(q) ?? false) ||
   342→        (p.excerptBn?.toLowerCase().includes(q) ?? false);
   343→      return matchCategory && matchSearch;
   344→    });
   345→  }, [posts, category, search]);
   346→
   347→  /* pagination — derived against the active filter key so "load more"
   348→     resets automatically whenever the category or search changes */
   349→  const filterKey = `${category}|${search}`;
   350→  const [pageState, setPageState] = useState({
   351→    key: filterKey,
   352→    visible: PAGE_SIZE_INITIAL,
   353→  });
   354→  const visible =
   355→    pageState.key === filterKey ? pageState.visible : PAGE_SIZE_INITIAL;
   356→  const showMore = () =>
   357→    setPageState({ key: filterKey, visible: visible + PAGE_SIZE_MORE });
   358→
   359→  const featured = posts.find((p) => p.featured) ?? posts[0];
   360→  /* The large featured card appears only on the unfiltered view; while
   361→     searching or inside a category every match lives in the grid (so the
   362→     visible count always matches the "Showing X articles" text). */
   363→  const showFeatured = !searching && category === "All" && Boolean(featured);
   364→  const rest = showFeatured
   365→    ? filtered.filter((p) => p.slug !== featured.slug)
   366→    : filtered;
   367→  const shown = rest.slice(0, visible);
   368→  const hasMore = rest.length > visible;
   369→
   370→  /* stats strip — computed from the live bundle */
   371→  const stats = useMemo(
   372→    () => ({
   373→      guides: posts.length,
   374→      topics: new Set(posts.map((p) => p.category)).size,
   375→      authors: new Set(posts.map((p) => p.author)).size,
   376→    }),
   377→    [posts]
   378→  );
   379→
   380→  /* sidebar "latest articles" — 5 most recent by publish date */
   381→  const latestPosts = useMemo(
   382→    () => [...posts].sort((a, b) => b.date.localeCompare(a.date)).slice(0, 5),
   383→    [posts]
   384→  );
   385→
   386→  const resultCountText = t.searchResults.replace(
   387→    "{count}",
   388→    localizeCount(bn, rest.length)
   389→  );
   390→  const showingOfText = t.showingOf
   391→    .replace("{shown}", localizeCount(bn, shown.length))
   392→    .replace("{total}", localizeCount(bn, rest.length));
   393→
   394→  const clearFilters = () => {
   395→    setSearch("");
   396→    setCategory("All");
   397→  };
   398→
   399→  const featuredTitle =
   400→    bn && featured?.titleBn ? featured.titleBn : featured?.title ?? "";
   401→
   402→  return (
   403→    <>
   404→      <PageHeader
   405→        eyebrow={t.eyebrow}
   406→        title={t.title}
   407→        description={t.description}
   408→        breadcrumb={[{ label: d.chrome.nav.home, to: "/" }, { label: d.chrome.nav.blog }]}
   409→      />
   410→
   411→      {/* Stats strip — floats over the hero's lower edge */}
   412→      <section className="relative z-10 mx-auto -mt-9 max-w-4xl px-4 sm:-mt-12 sm:px-6">
   413→        <Reveal>
   414→          <dl className="grid grid-cols-3 divide-x divide-blue-100 rounded-3xl border border-blue-100/80 bg-white/95 p-4 shadow-[0_18px_50px_-24px_rgba(0,32,96,0.28)] backdrop-blur sm:p-6">
   415→            <div className="flex flex-col items-center gap-1 px-2 text-center">
   416→              <dt className="flex items-center gap-1.5 text-[11px] font-medium uppercase tracking-wider text-slate-400">

   425→              <dt className="flex items-center gap-1.5 text-[11px] font-medium uppercase tracking-wider text-slate-400">

   832→
   833→function ReadingProgress() {
   834→  const [progress, setProgress] = useState(0);
   835→
   836→  useEffect(() => {
   837→    let raf = 0;
   838→    const update = () => {
   839→      raf = 0;
   840→      const el = document.documentElement;
   841→      const total = el.scrollHeight - el.clientHeight;
   842→      setProgress(total > 0 ? Math.min(1, el.scrollTop / total) : 0);
   843→    };
   844→    const onScroll = () => {
   845→      if (!raf) raf = requestAnimationFrame(update);
   846→    };
   847→    update();
   848→    window.addEventListener("scroll", onScroll, { passive: true });
   849→    window.addEventListener("resize", onScroll, { passive: true });
   850→    return () => {
   851→      window.removeEventListener("scroll", onScroll);
   852→      window.removeEventListener("resize", onScroll);
   853→      if (raf) cancelAnimationFrame(raf);
   854→    };
   855→  }, []);
   856→
   857→  return (
   858→    <div
   859→      className="reading-progress"
   860→      style={{ transform: `scaleX(${progress})` }}
   861→      aria-hidden="true"
   862→    />
   863→  );
   864→}

   866→/* Active-section tracking for the article table of contents.
   867→   `routeKey` (the post slug) re-runs the observer when navigating
   868→   between articles that happen to have the same section count; the
   869→   active index is DERIVED against the key so it resets on navigation
   870→   without setState-in-effect. */
   871→function useActiveSection(count: number, routeKey: string) {
   872→  const [state, setState] = useState({ key: routeKey, active: 0 });
   873→  const active = state.key === routeKey ? state.active : 0;
   874→
   875→  useEffect(() => {
   876→    if (count < 2) return;
   877→    const headings = Array.from({ length: count }, (_, i) =>
   878→      document.getElementById(`section-${i}`)
   879→    );
   880→    const visible = headings.filter(Boolean) as HTMLElement[];
   881→    if (visible.length === 0) return;
   882→    const observer = new IntersectionObserver(
   883→      (entries) => {
   884→        entries.forEach((entry) => {
   885→          if (entry.isIntersecting) {
   886→            const idx = Number(entry.target.id.replace("section-", ""));
   887→            setState({ key: routeKey, active: idx });
   888→          }
   889→        });
   890→      },
   891→      { rootMargin: "-25% 0px -65% 0px", threshold: 0 }
   892→    );
   893→    visible.forEach((h) => observer.observe(h));
   894→    return () => observer.disconnect();
   895→  }, [count, routeKey]);
   896→  return [active, (i: number) => setState({ key: routeKey, active: i })] as const;
   897→}
