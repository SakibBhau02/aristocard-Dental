   227→export function AdminPage() {
   228→  usePageMeta({
   229→    title: "Admin — Staff Dashboard",
   230→    description: "Staff-only clinic management dashboard.",
   231→  });
   232→
   233→  /* noindex for this back-office page */
   234→  useEffect(() => {
   235→    const meta = document.createElement("meta");
   236→    meta.name = "robots";
   237→    meta.content = "noindex, nofollow";
   238→    meta.dataset.adminNoindex = "true";
   239→    document.head.appendChild(meta);
   240→    return () => {
   241→      meta.remove();
   242→    };
   243→  }, []);
   244→
   245→  const [authState, setAuthState] = useState<"checking" | "locked" | "open">(
   246→    "checking"
   247→  );
   248→
   249→  /* Restore an existing cookie session (and clear any legacy
   250→     pre-session-era key) — the passcode is never re-sent or stored. */
   251→  useEffect(() => {
   252→    try {
   253→      sessionStorage.removeItem(LEGACY_SESSION_KEY);
   254→    } catch {
   255→      /* ignore */
   256→    }
   257→    let cancelled = false;
   258→    (async () => {
   259→      try {
   260→        const res = await fetch("/api/admin/session");
   261→        if (!cancelled) setAuthState(res.ok ? "open" : "locked");
   262→      } catch {
   263→        if (!cancelled) setAuthState("locked");
   264→      }
   265→    })();
   266→    return () => {
   267→      cancelled = true;
   268→    };
   269→  }, []);
   270→
   271→  const handleLogout = useCallback(async () => {
   272→    try {
   273→      await fetch("/api/admin/logout", { method: "POST" });
   274→    } catch {
   275→      /* clearing the local state matters more than the network call */
   276→    }
   277→    setAuthState("locked");
   278→  }, []);
   279→
   280→  if (authState === "checking") return <AdminBoot />;
   281→
   282→  return authState === "open" ? (
   283→    <AdminShell onLogout={handleLogout} />
   284→  ) : (
   285→    <AdminGate onUnlock={() => setAuthState("open")} />
   286→  );
   287→}

   312→/* ------------------------------------------------------------------ */
   313→/*  Passcode gate (visuals polished; auth logic unchanged)              */
   314→/* ------------------------------------------------------------------ */
   315→
   316→function AdminGate({ onUnlock }: { onUnlock: () => void }) {
   317→  const [value, setValue] = useState("");
   318→  const [error, setError] = useState<string | null>(null);
   319→  const [checking, setChecking] = useState(false);
   320→
   321→  async function handleSubmit(e: React.FormEvent) {
   322→    e.preventDefault();
   323→    if (checking) return;
   324→    setError(null);
   325→    setChecking(true);
   326→    try {
   327→      /* The passcode travels once, in this request body — the server
   328→         responds with an httpOnly session cookie that does the rest. */
   329→      const res = await fetch("/api/admin/login", {
   330→        method: "POST",
   331→        headers: { "Content-Type": "application/json" },
   332→        body: JSON.stringify({ code: value.trim() }),
   333→      });
   334→      if (res.status === 401) {
   335→        setError("Incorrect passcode. Please try again.");
   336→        return;
   337→      }
   338→      if (res.status === 429) {
   339→        setError("Too many attempts. Please wait a few minutes and try again.");
   340→        return;
   341→      }
   342→      if (!res.ok) throw new Error("Server error");
   343→      onUnlock();
   344→    } catch {
   345→      setError("Could not reach the server. Please try again.");
   346→    } finally {
   347→      setChecking(false);
   348→    }
   349→  }
   350→
   351→  return (
   352→    <div className="relative flex flex-1 items-center justify-center overflow-hidden bg-gradient-to-b from-brand-navy via-[#022450] to-[#011a34] px-4 py-20">
   353→      <div className="dot-grid-light pointer-events-none absolute inset-0 opacity-30" aria-hidden="true" />
   354→      <div className="pointer-events-none absolute -top-32 left-1/2 h-80 w-80 -translate-x-1/2 rounded-full bg-brand-blue/30 blur-3xl" aria-hidden="true" />
   355→      <div className="pointer-events-none absolute -bottom-40 right-[12%] h-72 w-72 rounded-full bg-amber-400/10 blur-3xl" aria-hidden="true" />
   356→
   357→      <motion.div
   358→        initial={{ opacity: 0, y: 24 }}
   359→        animate={{ opacity: 1, y: 0 }}
   360→        transition={{ duration: 0.5, ease: [0.22, 1, 0.36, 1] }}
   361→        className="relative w-full max-w-md"
   362→      >
   363→        <div className="mb-7 text-center">
   364→          <p className="font-heading text-xl font-bold tracking-wide text-white">
   365→            Aristocrat Dental Care
   366→          </p>
   367→          <p className="mt-1 text-[11px] font-semibold uppercase tracking-[0.24em] text-amber-300/90">
   368→            Admin Panel
   369→          </p>
   370→        </div>
   371→
   372→        <form
   373→          onSubmit={handleSubmit}
   374→          className="card-shine rounded-[2rem] border border-white/40 bg-white p-8 shadow-[0_32px_80px_-24px_rgba(0,10,40,0.7)] sm:p-10"
   375→        >
   376→          <div className="flex items-center justify-center">
   377→            <span className="inline-flex h-16 w-16 items-center justify-center rounded-3xl bg-gradient-to-br from-brand-navy to-brand-blue text-white shadow-lg shadow-navy-900/30 ring-1 ring-blue-900/20">
   378→              <Lock className="h-7 w-7" aria-hidden="true" />
   379→            </span>
   380→          </div>
   381→          <h1 className="font-heading mt-6 text-center text-2xl font-bold text-brand-navy">
   382→            Staff Access Only
   383→          </h1>
   384→          <p className="mt-2 text-center text-sm leading-relaxed text-slate-500">
   385→            Enter the clinic admin passcode to manage appointments and site
   386→            content. This area is for Aristocrat Dental Care staff.
   387→          </p>
   388→
   389→          <label htmlFor="admin-code" className="sr-only">
   390→            Admin passcode
   391→          </label>
   392→          <input
   393→            id="admin-code"
   394→            type="password"
   395→            autoComplete="current-password"
   396→            value={value}
   397→            onChange={(e) => setValue(e.target.value)}
   398→            placeholder="Admin passcode"
   399→            className="mt-7 h-13 w-full rounded-2xl border border-blue-100 bg-ice px-5 text-sm text-brand-navy placeholder:text-slate-400 focus:border-brand-blue focus:outline-none focus:ring-2 focus:ring-brand-blue/25"
   400→          />
   401→
   402→          {error && (
   403→            <p className="mt-3 rounded-xl bg-rose-50 px-4 py-2.5 text-xs font-medium text-rose-600 ring-1 ring-rose-100">
   404→              {error}
   405→            </p>
   406→          )}
   407→
   408→          <Button
   409→            type="submit"
   410→            disabled={checking || value.trim().length === 0}
   411→            className="mt-5 h-12 w-full rounded-2xl bg-brand-navy text-sm font-bold text-white shadow-lg shadow-navy-900/25 hover:bg-brand-blue disabled:opacity-60"
   412→          >
   413→            {checking ? (
   414→              <Loader2 className="h-4.5 w-4.5 animate-spin" aria-hidden="true" />
   415→            ) : (
   416→              <ShieldCheck className="h-4.5 w-4.5" aria-hidden="true" />
   417→            )}
   418→            Unlock Dashboard
   419→          </Button>
   420→
   421→          <p className="mt-5 text-center text-[11px] leading-relaxed text-slate-400">
   422→            You'll stay signed in for 8 hours on this device. Demo passcode:{" "}
   423→            <code className="rounded bg-slate-100 px-1.5 py-0.5 font-mono text-[10px] text-slate-600">
   424→              aristocrat-2026
   425→            </code>
   426→          </p>
   427→        </form>
   428→      </motion.div>
   429→    </div>
   430→  );
   431→}

   457→/* Resolve the active module's header metadata (title / description /
   458→   icon / group label). Cheap lookup — no memoization needed. */
   459→function navMeta(view: AdminView) {
   460→  for (const group of NAV_GROUPS) {
   461→    const item = group.items.find((i) => i.id === view);
   462→    if (item) return { ...item, group: group.label };
   463→  }
   464→  return { ...NAV_GROUPS[0].items[0], group: NAV_GROUPS[0].label };
   465→}

   468→/*  AdminShell — sidebar + content area + mobile Sheet                  */

   470→
   471→function AdminShell({ onLogout }: { onLogout: () => void }) {
   472→  const [view, setView] = useState<AdminView>("overview");
   473→  const [sessionExpired, setSessionExpired] = useState(false);
   474→  const [navOpen, setNavOpen] = useState(false);
   475→  const [badges, setBadges] = useState<BadgeState>({
   476→    appointments: null,
   477→    reviews: null,
   478→    messages: null,
   479→  });
   480→
   481→  const handleExpired = useCallback(() => setSessionExpired(true), []);
   482→  const setAppointmentsBadge = useCallback(
   483→    (n: number) => setBadges((b) => ({ ...b, appointments: n })),
   484→    []
   485→  );
   486→  const setReviewsBadge = useCallback(
   487→    (n: number) => setBadges((b) => ({ ...b, reviews: n })),
   488→    []
   489→  );
   490→  const setMessagesBadge = useCallback(
   491→    (n: number) => setBadges((b) => ({ ...b, messages: n })),
   492→    []
   493→  );
   494→  const syncBadges = useCallback(
   495→    (appointments: number, reviews: number, messages: number) =>
   496→      setBadges({ appointments, reviews, messages }),
   497→    []
   498→  );
   499→
   500→  /* One light call carries every sidebar badge count before the staff
   501→     member opens each view. (Cookie auth — no header.) */
   502→  useEffect(() => {
   503→    let cancelled = false;
   504→    (async () => {
   505→      try {
   506→        const res = await fetch("/api/admin/overview");
   507→        if (res.status === 401) {
   508→          setSessionExpired(true);
   509→          return;
   510→        }
   511→        if (!res.ok) return;
   512→        const data = await res.json();
   513→        if (cancelled || !data) return;
   514→        setBadges({
   515→          appointments: data.appointmentStats?.pending ?? 0,
   516→          reviews: data.reviewStats?.pendingApproval ?? 0,
   517→          messages: data.messageStats?.unhandled ?? 0,
   518→        });
   519→      } catch {
   520→        /* badges are best-effort */
   521→      }
   522→    })();
   523→    return () => {
   524→      cancelled = true;
   525→    };
   526→  }, []);
   527→
   528→  const navigate = useCallback((next: AdminView) => {
   529→    setView(next);
   530→    setNavOpen(false);
   531→    window.scrollTo({ top: 0 });
   532→  }, []);
   533→
   534→  const meta = navMeta(view);
   535→
   536→  if (sessionExpired) return <SessionExpiredCard onLogout={onLogout} />;
   537→
   538→  let panel: React.ReactNode = null;
   539→  switch (view) {
   540→    case "overview":
   541→      panel = (
   542→        <OverviewPanel
   543→          onSessionExpired={handleExpired}
   544→          onNavigate={navigate}
   545→          onBadgeCounts={syncBadges}
   546→        />
   547→      );
   548→      break;
   549→    case "appointments":
   550→      panel = (
   551→        <AppointmentsPanel
   552→          onSessionExpired={handleExpired}
   553→          onPendingChange={setAppointmentsBadge}
   554→        />
   555→      );
   556→      break;
   557→    case "reviews":
   558→      panel = (
   559→        <ReviewsPanel
   560→          onSessionExpired={handleExpired}
   561→          onPendingChange={setReviewsBadge}
   562→        />
   563→      );
   564→      break;
   565→    case "messages":
   566→      panel = (
   567→        <MessagesPanel
   568→          onSessionExpired={handleExpired}
   569→          onUnhandledChange={setMessagesBadge}
   570→        />
   571→      );
   572→      break;
   573→    case "subscribers":
   574→      panel = <SubscribersPanel onSessionExpired={handleExpired} />;
   575→      break;
   576→    case "patients":
   577→      panel = <PatientsPanel onSessionExpired={handleExpired} />;
   578→      break;
   579→    case "doctors":
   580→      panel = <DoctorsPanel onSessionExpired={handleExpired} />;
   581→      break;
   582→    case "blog":
   583→      panel = <BlogPanel onSessionExpired={handleExpired} />;
   584→      break;
   585→    case "faqs":
   586→      panel = <FaqsPanel onSessionExpired={handleExpired} />;
   587→      break;
   588→    case "cases":
   589→      panel = <CasesPanel onSessionExpired={handleExpired} />;
   590→      break;
   591→    case "services":
   592→      panel = <ServicesPanel onSessionExpired={handleExpired} />;
   593→      break;
   594→    case "homepage":
   595→      panel = <HomepagePanel onSessionExpired={handleExpired} />;
   596→      break;
   597→    case "about":
   598→      panel = <AboutPanel onSessionExpired={handleExpired} />;
   599→      break;
   600→    case "hero":
   601→      panel = <HeroPanel onSessionExpired={handleExpired} />;
   602→      break;
