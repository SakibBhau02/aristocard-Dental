   260→function toIsoDate(d: Date): string {
   261→  const p = (n: number) => String(n).padStart(2, "0");
   262→  return `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())}`;
   263→}
   264→
   265→export function localToday(): string {
   266→  return toIsoDate(new Date());

   269→function addDaysIso(iso: string, days: number): string {
   270→  const [y, m, d] = iso.split("-").map(Number);
   271→  return toIsoDate(new Date(y, m - 1, d + days));
   272→}

   293→/** "Oct 12, 2026" from an ISO date or full timestamp. */
   294→export function prettyDay(iso: string): string {
   295→  const d = new Date(iso);
   296→  if (Number.isNaN(d.getTime())) return iso;
   297→  return d.toLocaleDateString("en-US", {
   298→    month: "short",
   299→    day: "numeric",
   300→    year: "numeric",
   301→  });
   302→}
   303→
   304→/** "Oct 12, 2026 · 4:26 PM" from an ISO timestamp. */
   305→export function prettyStamp(iso: string): string {
   306→  const d = new Date(iso);
   307→  if (Number.isNaN(d.getTime())) return iso;
   308→  return `${prettyDay(iso)} · ${d.toLocaleTimeString("en-US", {
   309→    hour: "numeric",
   310→    minute: "2-digit",
   311→  })}`;
   312→}

   314→/** "3 days ago" style relative label (best-effort, local clock). */
   315→export function relativeTime(iso: string): string {
   316→  const then = new Date(iso).getTime();
   317→  if (Number.isNaN(then)) return "";
   318→  const mins = Math.round((Date.now() - then) / 60_000);
   319→  if (mins < 1) return "just now";
   320→  if (mins < 60) return `${mins} min ago`;
   321→  const hours = Math.round(mins / 60);
   322→  if (hours < 24) return `${hours} hr ago`;
   323→  const days = Math.round(hours / 24);
   324→  if (days < 7) return `${days} day${days === 1 ? "" : "s"} ago`;
   325→  const weeks = Math.round(days / 7);
   326→  if (weeks < 5) return `${weeks} wk ago`;
   327→  const months = Math.round(days / 30);
   328→  if (months < 12) return `${months} mo ago`;
   329→  return `${Math.round(months / 12)} yr ago`;
   330→}

   332→/** "Mon 29" chart label from a YYYY-MM-DD date. */
   333→export function chartDayLabel(iso: string): string {
   334→  const [y, m, d] = iso.split("-").map(Number);
   335→  if (!y || !m || !d) return iso;
   336→  const dt = new Date(Date.UTC(y, m - 1, d));
   337→  const weekday = dt.toLocaleDateString("en-US", {
   338→    timeZone: "UTC",
   339→    weekday: "short",
   340→  });
   341→  return `${weekday} ${d}`;
   342→}

   344→/** Avatar initials, e.g. "Demo Patient" → "DP". */
   345→export function initialsOf(name: string): string {
   346→  const parts = name.trim().split(/\s+/).filter(Boolean);
   347→  if (parts.length === 0) return "?";
   348→  return parts
   349→    .slice(0, 2)
   350→    .map((p) => p.charAt(0).toUpperCase())
   351→    .join("");

   374→/** Shared CSV download — the filename comes from Content-Disposition. */
   375→export async function downloadCsvFile(
   376→  path: string,
   377→  fallbackName: string
   378→): Promise<"ok" | "unauthorized" | "error"> {
   379→  try {
   380→    const res = await fetch(path);
   381→    if (res.status === 401) return "unauthorized";
   382→    if (!res.ok) return "error";
   383→    const blob = await res.blob();
   384→    const disposition = res.headers.get("Content-Disposition") ?? "";
   385→    const filename =
   386→      /filename="([^"]+)"/.exec(disposition)?.[1] ?? fallbackName;
   387→    const url = URL.createObjectURL(blob);
   388→    const a = document.createElement("a");
   389→    a.href = url;
   390→    a.download = filename;
   391→    document.body.appendChild(a);
   392→    a.click();
   393→    a.remove();
   394→    URL.revokeObjectURL(url);
   395→    return "ok";
   396→  } catch {
   397→    return "error";
   398→  }
   399→}

   496→/* Panels fire several parallel requests; collapse 403 notices so a
   497→   mid-session role change doesn't spam the toaster. */
   498→let lastForbiddenNoticeAt = 0;
   499→
   500→function notifyForbidden(
   501→  fail: (title: string, description?: string) => void,
   502→  message: string
   503→) {
   504→  const now = Date.now();
   505→  if (now - lastForbiddenNoticeAt < 15_000) return;
   506→  lastForbiddenNoticeAt = now;
   507→  fail("Not allowed", message);
   508→}
   509→
   510→export function useAdminFetch(onUnauthorized: () => void) {
   511→  const [loading, setLoading] = useState(false);
   512→  const [error, setError] = useState<string | null>(null);
   513→  const onAuthRef = useRef(onUnauthorized);
   514→  onAuthRef.current = onUnauthorized;
   515→  const { fail: toastFail } = useAdminToast();
   516→  const failRef = useRef(toastFail);
   517→  failRef.current = toastFail;
   518→
   519→  const handle401 = useCallback((res: Response): boolean => {
   520→    if (res.status === 401) {
   521→      onAuthRef.current();
   522→      return true;
   523→    }
   524→    return false;
   525→  }, []);
   526→
   527→  const handle403 = useCallback((res: Response, message: string) => {
   528→    if (res.status === 403) {
   529→      notifyForbidden(failRef.current, message);
   530→      return true;
   531→    }
   532→    return false;
   533→  }, []);
   534→
   535→  const get = useCallback(
   536→    async <T,>(path: string): Promise<T | null> => {
   537→      setLoading(true);
   538→      setError(null);
   539→      try {
   540→        const res = await fetch(path);
   541→        if (handle401(res)) return null;
   542→        const data = await res.json().catch(() => null);
   543→        if (!res.ok) {
   544→          const message =
   545→            (data as { error?: string } | null)?.error ?? `Request failed (${res.status}).`;
   546→          handle403(res, message);
   547→          throw new Error(message);
   548→        }
   549→        return data as T;
   550→      } catch (err) {
   551→        setError(err instanceof Error ? err.message : "Request failed.");
   552→        return null;
   553→      } finally {
   554→        setLoading(false);
   555→      }
   556→    },
   557→    [handle401, handle403]
   558→  );
   559→
   560→  const post = useCallback(
   561→    async <T,>(
   562→      path: string,
   563→      payload: unknown
   564→    ): Promise<{ ok: boolean; data: T | null; status: number }> => {
   565→      setLoading(true);
   566→      setError(null);
   567→      try {
   568→        const res = await fetch(path, {
   569→          method: "POST",
   570→          headers: { "Content-Type": "application/json" },
   571→          body: JSON.stringify(payload ?? {}),
   572→        });
   573→        if (handle401(res)) return { ok: false, data: null, status: 401 };
   574→        const data = await res.json().catch(() => null);
   575→        if (!res.ok) {
   576→          const message =
   577→            (data as { error?: string } | null)?.error ?? `Request failed (${res.status}).`;
   578→          handle403(res, message);
   579→          setError(message);
   580→          return { ok: false, data: data as T | null, status: res.status };
   581→        }
   582→        return { ok: true, data: data as T, status: res.status };
   583→      } catch (err) {
   584→        setError(err instanceof Error ? err.message : "Request failed.");
   585→        return { ok: false, data: null, status: 0 };
   586→      } finally {
   587→        setLoading(false);
   588→      }
   589→    },
   590→    [handle401, handle403]
   591→  );
   592→
   593→  const del = useCallback(
   594→    async <T,>(path: string): Promise<{ ok: boolean; data: T | null; status: number }> => {
   595→      setLoading(true);
   596→      setError(null);
   597→      try {
   598→        const res = await fetch(path, { method: "DELETE" });
   599→        if (handle401(res)) return { ok: false, data: null, status: 401 };
   600→        const data = await res.json().catch(() => null);
   601→        if (!res.ok) {
   602→          const message =
   603→            (data as { error?: string } | null)?.error ?? `Request failed (${res.status}).`;
   604→          handle403(res, message);
   605→          setError(message);
   606→          return { ok: false, data: data as T | null, status: res.status };
   607→        }
   608→        return { ok: true, data: data as T, status: res.status };
   609→      } catch (err) {
   610→        setError(err instanceof Error ? err.message : "Request failed.");
   611→        return { ok: false, data: null, status: 0 };
   612→      } finally {

   620→
   621→  return { get, post, del, loading, error, handle401, clearError };

   831→export function ImageUploadField({
   832→  value,
   833→  onChange,
   834→  label,
   835→  hint,
   836→  onUnauthorized,
   837→}: {
   838→  value: string | undefined;
   839→  onChange: (url: string | undefined) => void;
   840→  label?: string;
   841→  hint?: string;
   842→  onUnauthorized?: () => void;
   843→}) {
   844→  const inputRef = useRef<HTMLInputElement | null>(null);
   845→  const [uploading, setUploading] = useState(false);
   846→  const { fail } = useAdminToast();
   847→
   848→  async function uploadFile(file: File) {
   849→    if (uploading) return;
   850→    setUploading(true);
   851→    try {
   852→      const body = new FormData();
   853→      body.append("file", file);
   854→      const res = await fetch("/api/admin/upload", { method: "POST", body });
   855→      const data = (await res.json().catch(() => null)) as
   856→        | { url?: string; error?: string }
   857→        | null;
   858→      if (res.status === 401) {
   859→        onUnauthorized?.();
   860→        fail("Session expired", "Please sign in again and retry the upload.");
   861→        return;
   862→      }
   863→      if (!res.ok || !data?.url) {
   864→        throw new Error(data?.error ?? "The upload was rejected by the server.");
   865→      }
   866→      onChange(data.url);
   867→    } catch (err) {
   868→      fail("Upload failed", err instanceof Error ? err.message : "Please try again.");
   869→    } finally {
   870→      setUploading(false);
   871→      if (inputRef.current) inputRef.current.value = "";
   872→    }
   873→  }
   874→
   875→  return (
   876→    <div className="min-w-0">
   877→      {label ? (

   892→                src={value}
   893→                alt="Selected image preview"
   894→                fill
   895→                sizes="96px"
   896→                className="object-cover"
   897→              />
   898→            ) : (
   899→              <div className="flex h-full w-full items-center justify-center border-2 border-dashed border-blue-200 text-slate-300">
   900→                <Upload className="h-6 w-6" aria-hidden="true" />
   901→              </div>
   902→            )}
   903→          </div>
   904→          {value ? (
   905→            <button
   906→              type="button"
   907→              onClick={() => onChange(undefined)}
   908→              title="Remove image"
   909→              aria-label="Remove image"
   910→              className="absolute -right-2 -top-2 inline-flex h-6 w-6 items-center justify-center rounded-full bg-white text-rose-600 shadow ring-1 ring-rose-200 transition hover:bg-rose-50"

   917→              <Loader2 className="h-6 w-6 animate-spin text-brand-blue" aria-label="Uploading" />
   918→            </div>
   919→          ) : null}
   920→        </div>
   921→
   922→        {/* Controls */}
   923→        <div className="min-w-0 flex-1 space-y-2">
   924→          <input
   925→            ref={inputRef}
   926→            type="file"
   927→            accept="image/png,image/jpeg,image/webp,image/gif"
   928→            className="hidden"
   929→            onChange={(e) => {
   930→              const file = e.target.files?.[0];
   931→              if (file) void uploadFile(file);
   932→            }}
   933→          />

   935→            <Button
   936→              type="button"
   937→              variant="outline"
   938→              size="sm"
   939→              disabled={uploading}
   940→              onClick={() => inputRef.current?.click()}
   941→              className="rounded-full border-blue-200 text-brand-navy hover:bg-blue-50"
   942→            >
   943→              {uploading ? (
   944→                <Loader2 className="h-4 w-4 animate-spin" aria-hidden="true" />
   945→              ) : (
   946→                <Upload className="h-4 w-4" aria-hidden="true" />
   947→              )}
   948→              {uploading ? "Uploading…" : "Upload image"}
   949→            </Button>
   950→          </div>
   951→          <Input
   952→            value={value ?? ""}
   953→            onChange={(e) => onChange(e.target.value || undefined)}
   954→            placeholder="/images/example.png — or an /uploads/… path"
   955→            aria-label="Image path"
